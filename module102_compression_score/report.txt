"""
Module 102 - Compression Score Report
"""

from datetime import datetime


class CompressionScoreReport:

    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        lines = []

        lines.append("=" * 55)
        lines.append("MODULE 102")
        lines.append("COMPOSITE COMPRESSION SCORE")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")

        lines.append("\n" + "-" * 45)
        lines.append("COMPOSITE SCORE")
        lines.append("-" * 45)
        lines.append(f"  Compression Score:  {statistics.compression_score:.2f} / 100")
        lines.append(f"  Regime:             {statistics.regime}")
        lines.append(f"  Detail:             {statistics.regime_detail}")

        if statistics.weights:
            lines.append("\n" + "-" * 45)
            lines.append("COMPONENT WEIGHTS")
            lines.append("-" * 45)
            for component, weight in statistics.weights.items():
                lines.append(f"  {component}: {weight:.2f}")

        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)

        return "\n".join(lines)

    @staticmethod
    def save(report: str, filepath: str):
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
