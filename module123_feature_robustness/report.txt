"""
Module 123 - Feature Robustness Report
"""

from datetime import datetime
import pandas as pd


class FeatureRobustnessReport:
    """Report generator for feature robustness results."""
    
    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        """Generate a formatted report."""
        lines = []
        
        lines.append("=" * 55)
        lines.append("MODULE 123")
        lines.append("FEATURE ROBUSTNESS")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")
        
        # Overview
        lines.append("\n" + "-" * 45)
        lines.append("OVERVIEW")
        lines.append("-" * 45)
        lines.append(f"  Total Features: {statistics.total_features}")
        lines.append(f"  PASS: {statistics.passed_count}")
        lines.append(f"  WARNING: {statistics.warning_count}")
        lines.append(f"  FAIL: {statistics.failed_count}")
        
        # Percentages
        lines.append("\n" + "-" * 45)
        lines.append("PERCENTAGES")
        lines.append("-" * 45)
        lines.append(f"  PASS: {statistics.passed_percentage:.1f}%")
        lines.append(f"  WARNING: {statistics.warning_percentage:.1f}%")
        lines.append(f"  FAIL: {statistics.failed_percentage:.1f}%")
        
        # Robustness Scores
        lines.append("\n" + "-" * 45)
        lines.append("ROBUSTNESS SCORES")
        lines.append("-" * 45)
        lines.append(f"  Average: {statistics.average_robustness:.3f}")
        lines.append(f"  Median: {statistics.median_robustness:.3f}")
        lines.append(f"  Best: {statistics.best_score:.3f}")
        lines.append(f"  Worst: {statistics.worst_score:.3f}")
        
        # Best and Worst Features
        if statistics.best_feature:
            lines.append("\n" + "-" * 45)
            lines.append("BEST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.best_feature}: {statistics.best_score:.3f}")
        
        if statistics.worst_feature:
            lines.append("\n" + "-" * 45)
            lines.append("WORST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.worst_feature}: {statistics.worst_score:.3f}")
        
        # Metrics
        lines.append("\n" + "-" * 45)
        lines.append("AVERAGE METRICS")
        lines.append("-" * 45)
        lines.append(f"  Mean Score: {statistics.average_mean_score:.3f}")
        lines.append(f"  Std Score: {statistics.average_std_score:.3f}")
        lines.append(f"  Consistency: {statistics.average_consistency:.3f}")
        
        # Interpretation
        lines.append("\n" + "-" * 45)
        lines.append("INTERPRETATION")
        lines.append("-" * 45)
        
        if statistics.average_robustness > 0.7:
            lines.append("  ✅ Features are highly robust across regimes")
        elif statistics.average_robustness > 0.5:
            lines.append("  ✓ Features show moderate robustness")
        else:
            lines.append("  ⚠ Features show low robustness across regimes")
        
        if statistics.average_consistency > 0.7:
            lines.append("  ✅ Features perform consistently across regimes")
        elif statistics.average_consistency > 0.5:
            lines.append("  ✓ Features show moderate consistency")
        else:
            lines.append("  ⚠ Features show inconsistent performance across regimes")
        
        # Validation
        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        
        # Footer
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)
        
        return "\n".join(lines)
    
    @staticmethod
    def generate_robustness_table(robustness_table: pd.DataFrame, top_n: int = 10) -> str:
        """Generate a robustness table report."""
        lines = []
        lines.append("\n" + "=" * 80)
        lines.append(f"TOP {top_n} FEATURES BY ROBUSTNESS")
        lines.append("=" * 80)
        
        top_features = robustness_table.head(top_n)
        
        lines.append(
            f"\n{'Feature':<20} {'Robustness':<12} {'Consistency':<14} {'Mean':<10} {'Std':<10} {'Decision':<12}"
        )
        lines.append("-" * 80)
        
        for _, row in top_features.iterrows():
            lines.append(
                f"{row['feature'][:19]:<20} "
                f"{row['robustness_score']:<12.3f} "
                f"{row['consistency_index']:<14.3f} "
                f"{row['mean_score']:<10.3f} "
                f"{row['std_score']:<10.3f} "
                f"{row['decision']:<12}"
            )
        
        lines.append("\n" + "=" * 80)
        return "\n".join(lines)
    
    @staticmethod
    def generate_regime_report(regime_scores: pd.DataFrame) -> str:
        """Generate a regime breakdown report."""
        lines = []
        lines.append("\n" + "=" * 60)
        lines.append("REGIME PERFORMANCE")
        lines.append("=" * 60)
        
        # Pivot table: feature x regime
        pivot = regime_scores.pivot_table(
            index='feature',
            columns='regime',
            values='score'
        )
        
        lines.append("\n" + pivot.round(4).to_string())
        lines.append("\n" + "=" * 60)
        return "\n".join(lines)
    
    @staticmethod
    def save(report: str, filepath: str):
        """Save report to file."""
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
