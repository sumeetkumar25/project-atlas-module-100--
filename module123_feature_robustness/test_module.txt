def create_test_data(n=1000, seed=42):
    """Create synthetic test data with regimes and feature robustness."""
    np.random.seed(seed)
    
    # Create price data with regimes
    close = pd.Series(
        100 + np.cumsum(np.random.randn(n) * 0.5)
    )
    high = close + np.abs(np.random.randn(n) * 1.0)
    low = close - np.abs(np.random.randn(n) * 1.0)
    volume = np.random.randn(n) * 100 + 1000
    
    # Create signal from price
    signal = close.diff().shift(1).fillna(0)
    
    # Target directly from signal (deterministic)
    target = (signal > 0).astype(int)
    
    # Feature 1: Perfect predictor (signal)
    perfect_feature = signal
    
    # Feature 2: Robust across all regimes (signal + small noise)
    robust_feature = signal + np.random.randn(n) * 0.10
    
    # Feature 3: Fragile - only predictive in first half
    fragile_feature = signal.copy()
    half = n // 2
    fragile_feature.iloc[:half] += np.random.randn(half) * 0.10
    fragile_feature.iloc[half:] = np.random.randn(n - half) * 0.5
    
    # Feature 4: Noise (not predictive)
    noise_feature = np.random.randn(n) * 0.5
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': high,
        'low': low,
        'close': close,
        'volume': volume,
        'perfect_feature': perfect_feature,
        'robust_feature': robust_feature,
        'fragile_feature': fragile_feature,
        'noise_feature': noise_feature,
        'future_breakout': target
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRobustnessEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'perfect_feature' in features
    assert 'robust_feature' in features
    assert 'fragile_feature' in features
    assert 'noise_feature' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True

def test_engine_robustness_metrics():
    """Test 7: Robustness metrics computation."""
    print("\n" + "=" * 60)
    print("TEST 7: Robustness Metrics")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRobustnessEngine()
    
    features = engine.get_feature_columns(df)
    y = df['future_breakout'].values
    
    regime_scores = engine.compute_regime_scores(df, features, y)
    metrics = engine.compute_robustness_metrics(regime_scores)
    
    # Debug output to understand feature rankings
    print("\n================ ROBUSTNESS TABLE ================\n")
    print(
        metrics[
            [
                "feature",
                "mean_score",
                "min_score",
                "max_score",
                "consistency_index",
                "robustness_score",
                "decision",
            ]
        ].sort_values("robustness_score", ascending=False)
    )
    
    print("\n================ REGIME SCORES ================\n")
    print(
        regime_scores.pivot_table(
            index="feature",
            columns="regime",
            values="score"
        ).round(3)
    )
    
    assert not metrics.empty
    assert 'robustness_score' in metrics.columns
    assert 'consistency_index' in metrics.columns
    assert 'decision' in metrics.columns
    
    # Perfect feature should have very high robustness (> 0.90)
    perfect_row = metrics[metrics['feature'] == 'perfect_feature']
    if not perfect_row.empty:
        assert perfect_row['robustness_score'].iloc[0] > 0.90
    
    # Noise feature should be classified as FAIL
    noise_row = metrics[metrics['feature'] == 'noise_feature']
    if not noise_row.empty:
        # Validate that the engine correctly classifies noise as FAIL
        assert noise_row['decision'].iloc[0] == "FAIL"
        # Validate that noise score is below the warning threshold
        assert noise_row['robustness_score'].iloc[0] < engine.warning_threshold
    
    print(f"  Robustness metrics: {len(metrics)} rows")
    print("  ✓ Robustness metrics test passed")
    return True


def test_engine_compute():

    result = engine.compute(df)
    
    assert not result.robustness_table.empty
    assert not result.regime_scores.empty

        len(result.passed_features)
        + len(result.warning_features)
        + len(result.failed_features)
        == result.summary_statistics["total_features"]
    )
    assert result.summary_statistics["average_robustness"] > 0
    
    # Perfect feature should be in PASS
    assert 'perfect_feature' in result.passed_features
    
    # Noise feature should be in FAIL
    assert 'noise_feature' in result.failed_features
    
    print(f"  PASS: {len(result.passed_features)}")
    print(f"  WARNING: {len(result.warning_features)}")
    print(f"  FAIL: {len(result.failed_features)}")
    print(f"  Avg Robustness: {result.summary_statistics['average_robustness']:.3f}")
    print("  ✓ Full engine test passed")
    return True
