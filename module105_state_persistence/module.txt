# atlas/modules/module105_state_persistence/module.py
"""
Module 105
State Persistence Analysis

Analyzes how long market states persist before transitioning.

Pipeline: load states → detect sequences → aggregate statistics → classification → validation → report → export
"""

from typing import Optional, Dict, Any
import pandas as pd
import numpy as np

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import StatePersistenceEngine
from .statistics import StatePersistenceStatisticsEngine
from .validator import run_validation
from .report import StatePersistenceReport


class StatePersistenceResearch(BaseResearchModule):
    """Main research module for state persistence analysis."""
    
    MODULE_OUTPUT = "module105"
    METADATA = StatePersistenceStatisticsEngine.METADATA
    
    def __init__(self, registry: Optional[AtlasRegistry] = None):
        """
        Initialize the state persistence module.
        
        Args:
            registry: Optional Atlas registry
        """
        super().__init__(
            module_name="State Persistence Analysis",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = StatePersistenceEngine()
        self.stats_engine = StatePersistenceStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.state_report = None
    
    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("state_persistence", self._compute_persistence),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data from DataFrame."""
        # Check if we have states column (from Module 103)
        if "state" not in self.df.columns:
            raise ValueError("DataFrame must contain 'state' column (from Module 103)")
        
        return {
            "states": self.df["state"].values
        }
    
    def _compute_persistence(self) -> Dict[str, Any]:
        """Step 1: Compute state persistence."""
        self.logger.info("Step 1: Computing state persistence")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            states=prepared["states"]
        )
        
        self.logger.metric(
            "Total Sequences",
            str(self.result_data.total_sequences)
        )
        self.logger.metric(
            "States Present",
            f"{len(self.result_data.persistence_table)}/16"
        )
        self.logger.metric(
            "Average Duration",
            f"{self.result_data.average_duration:.2f} bars"
        )
        self.logger.metric(
            "Longest Duration",
            f"{self.result_data.longest_duration} bars"
        )
        self.logger.success("State persistence computed")
        
        # Generate state report
        self.state_report = StatePersistenceReport.generate_state_report(
            self.result_data.persistence_table
        )
        
        return {
            "total_sequences": self.result_data.total_sequences,
            "average_duration": self.result_data.average_duration,
            "longest_duration": self.result_data.longest_duration,
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Compute statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Persistent States",
            f"{self.statistics.persistent_states}/{self.statistics.total_states}"
        )
        self.logger.metric(
            "Transient States",
            f"{self.statistics.transient_states}/{self.statistics.total_states}"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Store persistence table
        self.result.artifacts["persistence_table"] = self.result_data.persistence_table
        
        # Store main report
        main_report = StatePersistenceReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["state_persistence_report"] = main_report
        
        # Store state report
        self.result.artifacts["state_report"] = self.state_report
        
        # Store detailed results as dict
        self.result.artifacts["detailed_results"] = {
            "persistence_table": self.result_data.persistence_table.to_dict(orient="records"),
            "average_duration": self.result_data.average_duration,
            "median_duration": self.result_data.median_duration,
            "longest_duration": self.result_data.longest_duration,
            "shortest_duration": self.result_data.shortest_duration,
            "total_sequences": self.result_data.total_sequences,
            "sample_size": self.result_data.sample_size
        }
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("State Persistence Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 105 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on DataFrame."""
        return super().run_dataframe(df, dataset_id, output_dir)
