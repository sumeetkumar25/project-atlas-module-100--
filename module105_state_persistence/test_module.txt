def create_test_data(n=1000, seed=42):
    """
    Create synthetic test data with state persistence.
    
    Args:
        n: Number of samples
        seed: Random seed for reproducibility
        
    Returns:
        DataFrame with states
    """
    np.random.seed(seed)
    
    # Create states with some persistence
    states = []
    current_state = np.random.randint(0, 16)
    
    for _ in range(n):
        # 70% chance to stay in same state, 30% to change
        if np.random.random() < 0.7:
            states.append(current_state)
        else:
            current_state = np.random.randint(0, 16)
            states.append(current_state)
    
    df = pd.DataFrame({
        "state": states
    })
    
    return df


def test_engine_detect_sequences():
    """Test 1: Sequence detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Sequence Detection")
    print("=" * 60)
    
    engine = StatePersistenceEngine()
    
    # Test with known sequence
    states = np.array([4, 4, 4, 4, 7, 7, 2, 2, 2, 13, 13, 13, 13, 13, 5, 5, 5, 5, 5])
    
    sequence_df = engine.detect_sequences(states)
    
    # Check results
    assert len(sequence_df) == 5  # 5 sequences
    assert sequence_df.iloc[0]["state"] == 4
    assert sequence_df.iloc[0]["duration"] == 4
    assert sequence_df.iloc[1]["state"] == 7
    assert sequence_df.iloc[1]["duration"] == 2
    assert sequence_df.iloc[2]["state"] == 2
    assert sequence_df.iloc[2]["duration"] == 3
    assert sequence_df.iloc[3]["state"] == 13
    assert sequence_df.iloc[3]["duration"] == 5
    assert sequence_df.iloc[4]["state"] == 5
    assert sequence_df.iloc[4]["duration"] == 5
    
    print("  Sequence detection results:")
    for _, row in sequence_df.iterrows():
        print(f"    State {row['state']}: {row['duration']} bars")
    print("  ✓ Sequence detection test passed")
    return True


def test_engine_aggregate_statistics():
    """Test 2: Aggregate statistics."""
    print("\n" + "=" * 60)
    print("TEST 2: Aggregate Statistics")
    print("=" * 60)
    
    engine = StatePersistenceEngine()
    
    # Create test sequences
    states = np.array([4, 4, 4, 4, 7, 7, 2, 2, 2, 13, 13, 13, 13, 13, 5, 5, 5, 5, 5])
    sequence_df = engine.detect_sequences(states)
    summary = engine.aggregate_statistics(sequence_df)
    
    # Test specific assertions
    assert summary.loc[summary.state == 4, "mean_duration"].iloc[0] == 4
    assert summary.loc[summary.state == 7, "mean_duration"].iloc[0] == 2
    assert summary.loc[summary.state == 2, "mean_duration"].iloc[0] == 3
    assert summary.loc[summary.state == 13, "mean_duration"].iloc[0] == 5
    assert summary.loc[summary.state == 5, "mean_duration"].iloc[0] == 5
    
    # Check all states are present
    expected_states = [2, 4, 5, 7, 13]
    assert set(summary["state"].values) == set(expected_states)
    
    print("  Aggregate statistics:")
    for _, row in summary.iterrows():
        print(f"    State {row['state']}: mean={row['mean_duration']:.2f}, "
              f"sequences={row['sequences']}, max={row['max_duration']}")
    print("  ✓ Aggregate statistics test passed")
    return True


def test_engine_compute():
    """Test 3: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Full Engine Computation")
    print("=" * 60)
    
    engine = StatePersistenceEngine()
    
    # Create test data with known pattern
    states = np.array([4, 4, 4, 4, 7, 7, 2, 2, 2, 13, 13, 13, 13, 13])
    
    result = engine.compute(states)
    
    # Check results
    assert result.total_sequences == 4
    assert result.sample_size == 14
    assert result.average_duration == 3.5  # (4+2+3+5)/4
    assert result.median_duration == 3.5
    assert result.longest_duration == 5
    assert result.shortest_duration == 2
    assert not result.persistence_table.empty
    assert len(result.persistence_table) == 4  # 4 unique states
    
    print(f"  Total Sequences: {result.total_sequences}")
    print(f"  Average Duration: {result.average_duration:.2f}")
    print(f"  Longest Duration: {result.longest_duration}")
    print(f"  Shortest Duration: {result.shortest_duration}")
    print("  ✓ Full engine test passed")
    return True
