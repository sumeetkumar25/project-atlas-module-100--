"""
Module 122 - Feature Predictive Power Statistics
"""

from dataclasses import dataclass, asdict
import pandas as pd
from .engine import FeaturePredictivePowerResult


@dataclass
class FeaturePredictivePowerStatistics:
    """High-level statistics from feature predictive power analysis."""
    total_features: int
    passed_count: int
    warning_count: int
    failed_count: int
    passed_percentage: float
    warning_percentage: float
    failed_percentage: float
    average_predictive_score: float
    median_predictive_score: float
    min_predictive_score: float
    max_predictive_score: float
    best_feature: str
    worst_feature: str
    best_score: float
    worst_score: float
    average_mi: float
    average_roc_auc: float
    average_effect_size: float


class FeaturePredictivePowerStatisticsEngine:
    """Engine for generating high-level statistics."""
    
    METADATA = {
        "module": 122,
        "name": "Feature Predictive Power",
        "version": "1.0.0",
        "description": "Evaluates every feature individually against the prediction target"
    }
    
    def analyze(self, result: FeaturePredictivePowerResult) -> FeaturePredictivePowerStatistics:
        """Analyze results and generate high-level statistics."""
        return FeaturePredictivePowerStatistics(
            total_features=result.summary_statistics['total_features'],
            passed_count=result.summary_statistics['passed_count'],
            warning_count=result.summary_statistics['warning_count'],
            failed_count=result.summary_statistics['failed_count'],
            passed_percentage=result.summary_statistics['passed_percentage'],
            warning_percentage=result.summary_statistics['warning_percentage'],
            failed_percentage=result.summary_statistics['failed_percentage'],
            average_predictive_score=result.summary_statistics['average_predictive_score'],
            median_predictive_score=result.summary_statistics['median_predictive_score'],
            min_predictive_score=result.summary_statistics['min_predictive_score'],
            max_predictive_score=result.summary_statistics['max_predictive_score'],
            best_feature=result.summary_statistics['best_feature'],
            worst_feature=result.summary_statistics['worst_feature'],
            best_score=result.summary_statistics['best_score'],
            worst_score=result.summary_statistics['worst_score'],
            average_mi=result.summary_statistics['average_mi'],
            average_roc_auc=result.summary_statistics['average_roc_auc'],
            average_effect_size=result.summary_statistics['average_effect_size']
        )
    
    def summary_dataframe(self, stats: FeaturePredictivePowerStatistics) -> pd.DataFrame:
        """Convert statistics to DataFrame."""
        return pd.DataFrame([asdict(stats)])
    
    def summary_dict(self, stats: FeaturePredictivePowerStatistics) -> dict:
        """Convert statistics to dictionary."""
        return asdict(stats)
