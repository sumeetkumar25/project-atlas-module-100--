"""
Module 122 - Feature Predictive Power Validator
"""

import numpy as np
import pandas as pd
from .engine import FeaturePredictivePowerResult
from .statistics import FeaturePredictivePowerStatistics


class FeaturePredictivePowerValidator:
    """Validator for feature predictive power results."""
    
    def validate(
        self,
        result: FeaturePredictivePowerResult,
        statistics: FeaturePredictivePowerStatistics
    ) -> bool:
        """Validate results and statistics."""
        issues = []
        warnings = []
        
        # Check predictive table
        if result.predictive_table.empty:
            issues.append("Predictive table is empty")
        else:
            required_cols = ['feature', 'predictive_score', 'decision']
            missing = set(required_cols) - set(result.predictive_table.columns)
            if missing:
                issues.append(f"Missing columns in predictive table: {missing}")
            
            # Check scores are in [0, 1]
            if 'predictive_score' in result.predictive_table.columns:
                if not np.all((0 <= result.predictive_table['predictive_score']) & 
                            (result.predictive_table['predictive_score'] <= 1)):
                    issues.append("Predictive scores outside [0, 1] range")
            
            # Check ROC AUC in [0, 1]
            if 'roc_auc' in result.predictive_table.columns:
                if not np.all((0 <= result.predictive_table['roc_auc']) & 
                            (result.predictive_table['roc_auc'] <= 1)):
                    issues.append("ROC AUC values outside [0, 1] range")
            
            # Check MI >= 0
            if 'mutual_information' in result.predictive_table.columns:
                if not np.all(result.predictive_table['mutual_information'] >= 0):
                    issues.append("Mutual Information values are negative")
            
            # Check decisions are valid
            if 'decision' in result.predictive_table.columns:
                valid_decisions = {'PASS', 'WARNING', 'FAIL'}
                invalid = set(result.predictive_table['decision'].unique()) - valid_decisions
                if invalid:
                    issues.append(f"Invalid decisions found: {invalid}")
            
            # Check no NaNs
            if result.predictive_table.isna().any().any():
                issues.append("NaN values found in predictive table")
        
        # Check feature lists
        total_decision_count = len(result.passed_features) + len(result.warning_features) + len(result.failed_features)
        if total_decision_count != len(result.predictive_table):
            issues.append(
                f"Decision count mismatch: {total_decision_count} vs {len(result.predictive_table)}"
            )
        
        # Check no duplicates across decision lists
        all_decisions = set(result.passed_features) | set(result.warning_features) | set(result.failed_features)
        if len(all_decisions) != len(result.predictive_table):
            issues.append("Features appear in multiple decision lists")
        
        # Check statistics match result
        if statistics.total_features != result.summary_statistics['total_features']:
            issues.append(
                f"Total features mismatch: stats={statistics.total_features}, "
                f"result={result.summary_statistics['total_features']}"
            )
        
        if statistics.passed_count != len(result.passed_features):
            issues.append(
                f"Passed count mismatch: stats={statistics.passed_count}, "
                f"result={len(result.passed_features)}"
            )
        
        if statistics.warning_count != len(result.warning_features):
            issues.append(
                f"Warning count mismatch: stats={statistics.warning_count}, "
                f"result={len(result.warning_features)}"
            )
        
        if statistics.failed_count != len(result.failed_features):
            issues.append(
                f"Failed count mismatch: stats={statistics.failed_count}, "
                f"result={len(result.failed_features)}"
            )
        
        # Warnings (non-fatal)
        if statistics.passed_percentage < 20:
            warnings.append(f"Low pass percentage: {statistics.passed_percentage:.1f}%")
        
        if statistics.average_predictive_score < 0.4:
            warnings.append(f"Low average predictive score: {statistics.average_predictive_score:.3f}")
        
        if warnings:
            print("\n  WARNINGS:")
            for warning in warnings:
                print(f"    ⚠ {warning}")
        
        if issues:
            raise ValueError(
                f"Feature Predictive Power validation failed with {len(issues)} issue(s):\n" +
                "\n".join(f"  - {issue}" for issue in issues)
            )
        
        return True


def run_validation(result, statistics):
    """Run validation with user-friendly output."""
    validator = FeaturePredictivePowerValidator()
    
    print("\n" + "=" * 60)
    print("FEATURE PREDICTIVE POWER VALIDATION")
    print("=" * 60)
    
    try:
        validator.validate(result, statistics)
        print("\n  ✓ All validation checks passed")
        print("=" * 60)
        return True
    except ValueError as e:
        print(e)
        print("=" * 60)
        return False
