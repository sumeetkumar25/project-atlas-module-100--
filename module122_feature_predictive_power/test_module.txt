def create_test_data(n=500, seed=42):
    """Create synthetic test data with various predictive power levels."""
    np.random.seed(seed)
    
    # Strong predictive feature
    strong = np.random.randn(n) * 0.5 + 0
    strong_target = 1 / (1 + np.exp(-strong))
    strong_label = (strong_target > 0.5).astype(int)
    
    # Moderate predictive feature
    moderate = np.random.randn(n) * 0.5 + 0
    moderate_target = 1 / (1 + np.exp(-moderate * 0.3))
    moderate_label = (moderate_target > 0.5).astype(int)
    
    # Weak predictive feature
    weak = np.random.randn(n) * 0.5 + 0
    weak_target = 1 / (1 + np.exp(-weak * 0.1))
    weak_label = (weak_target > 0.5).astype(int)
    
    # Noise feature (no predictive power)
    noise = np.random.randn(n)
    noise_label = np.random.randint(0, 2, n)
    
    # Target
    target = strong_label
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'strong_feature': strong,
        'moderate_feature': moderate,
        'weak_feature': weak,
        'noise_feature': noise,
        'future_breakout': target
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeaturePredictivePowerEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'strong_feature' in features
    assert 'moderate_feature' in features
    assert 'weak_feature' in features
    assert 'noise_feature' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_pearson():
    """Test 2: Pearson correlation."""
    print("\n" + "=" * 60)
    print("TEST 2: Pearson Correlation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeaturePredictivePowerEngine()
    y = df['future_breakout'].values
    
    strong_corr = engine.compute_pearson(df['strong_feature'].values, y)
    noise_corr = engine.compute_pearson(df['noise_feature'].values, y)
    
    assert abs(strong_corr) > abs(noise_corr)
    assert -1 <= strong_corr <= 1
    
    print(f"  Strong feature correlation: {strong_corr:.4f}")
    print("  ✓ Pearson test passed")
    return True

def test_engine_compute():
    """Test 10: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 10: Full Engine Computation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeaturePredictivePowerEngine()
    
    result = engine.compute(df)
    
    assert not result.predictive_table.empty
    assert len(result.passed_features) >= 1
    assert (
        len(result.passed_features)
        + len(result.warning_features)
        + len(result.failed_features)
        == result.summary_statistics["total_features"]
    )
    assert result.summary_statistics["average_predictive_score"] > 0
    
    print(f"  PASS: {len(result.passed_features)}")
    print(f"  WARNING: {len(result.warning_features)}")
    print(f"  FAIL: {len(result.failed_features)}")
    print(f"  Avg Predictive Score: {result.summary_statistics['average_predictive_score']:.3f}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 11: Statistics engine."""
    print("\n" + "=" * 60)
    print("TEST 11: Statistics Engine")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeaturePredictivePowerEngine()
    stats_engine = FeaturePredictivePowerStatisticsEngine()
    
    result = engine.compute(df)
    stats = stats_engine.analyze(result)
    
    assert stats.total_features == result.summary_statistics['total_features']
    assert stats.passed_count == len(result.passed_features)
    assert stats.failed_count == len(result.failed_features)
    assert stats.best_feature is not None
    
    print(f"  Total: {stats.total_features}")
    print(f"  PASS: {stats.passed_count}")
    print(f"  FAIL: {stats.failed_count}")
    print("  ✓ Statistics test passed")
    return True
