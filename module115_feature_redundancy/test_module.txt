import sys

import numpy as np
import pandas as pd

from atlas.modules.module115_feature_redundancy.engine import FeatureRedundancyEngine
from atlas.modules.module115_feature_redundancy.statistics import FeatureRedundancyStatisticsEngine
from atlas.modules.module115_feature_redundancy.validator import run_validation

def create_test_data(n=500, seed=42):
    """Create synthetic test data with redundant features."""
    np.random.seed(seed)
    
    # Create base features
    x1 = np.random.randn(n)
    x2 = np.random.randn(n) * 0.5 + x1 * 0.5  # Correlated with x1
    x3 = np.random.randn(n) * 0.3 + x1 * 0.7  # Highly correlated with x1
    x4 = np.random.randn(n)  # Independent
    x5 = np.random.randn(n) * 0.5 + x4 * 0.5  # Correlated with x4
    
    # Create some non-linear features
    x6 = x1 ** 2 + np.random.randn(n) * 0.2  # Non-linear relationship
    x7 = np.sin(x1) + np.random.randn(n) * 0.2  # Non-linear relationship
    
    # Create target with some signal
    future_breakout = ((x1 > 0) & (x4 > 0)).astype(int)
    
    df = pd.DataFrame({
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'feature5': x5,
        'feature6': x6,
        'feature7': x7,
        'future_breakout': future_breakout
    })
    
    return df


def test_engine_features():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'feature1' in features
    assert 'feature2' in features
    assert 'feature3' in features
    assert 'feature4' in features
    assert 'feature5' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_pearson():
    """Test 2: Pearson correlation."""
    print("\n" + "=" * 60)
    print("TEST 2: Pearson Correlation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    pearson = engine.pearson_matrix(df, features)
    
    assert pearson.shape == (len(features), len(features))
    assert not pearson.isna().any().any()
    
    # feature1 and feature3 should be highly correlated
    assert abs(pearson.loc['feature1', 'feature3']) > 0.5
    
    print(f"  Pearson matrix shape: {pearson.shape}")
    print("  ✓ Pearson test passed")
    return True


def test_engine_mutual_information():
    """Test 3: Mutual Information."""
    print("\n" + "=" * 60)
    print("TEST 3: Mutual Information")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    mi = engine.mutual_information_matrix(df, features)
    
    assert mi.shape == (len(features), len(features))
    assert not mi.isna().any().any()
    assert np.all(mi >= 0)
    
    print(f"  MI matrix shape: {mi.shape}")
    print("  ✓ Mutual Information test passed")
    return True


def test_engine_distance_correlation():
    """Test 4: Distance correlation."""
    print("\n" + "=" * 60)
    print("TEST 4: Distance Correlation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    dc = engine.distance_correlation_matrix(df, features)
    
    assert dc.shape == (len(features), len(features))
    assert not dc.isna().any().any()
    assert np.all(dc >= 0)
    assert np.all(dc <= 1)
    
    print(f"  Distance correlation matrix shape: {dc.shape}")
    print("  ✓ Distance correlation test passed")
    return True


def test_engine_redundancy_score():
    """Test 5: Redundancy score computation."""
    print("\n" + "=" * 60)
    print("TEST 5: Redundancy Score")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    pearson = engine.pearson_matrix(df, features)
    spearman = engine.spearman_matrix(df, features)
    mi = engine.mutual_information_matrix(df, features)
    dc = engine.distance_correlation_matrix(df, features)
    
    score = engine.compute_redundancy_score(pearson, spearman, mi, dc, features)
    
    assert score.shape == (len(features), len(features))
    assert not score.isna().any().any()
    assert np.all(score >= 0)
    assert np.all(score <= 1)
    
    print(f"  Redundancy score matrix shape: {score.shape}")
    print("  ✓ Redundancy score test passed")
    return True


def test_engine_clustering():
    """Test 6: Feature clustering."""
    print("\n" + "=" * 60)
    print("TEST 6: Feature Clustering")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine(cluster_threshold=0.40)
    
    features = engine.get_feature_columns(df)
    pearson = engine.pearson_matrix(df, features)
    spearman = engine.spearman_matrix(df, features)
    mi = engine.mutual_information_matrix(df, features)
    dc = engine.distance_correlation_matrix(df, features)
    score = engine.compute_redundancy_score(pearson, spearman, mi, dc, features)
    
    cluster_df, cluster_dict = engine.cluster_features(score, features)
    
    assert not cluster_df.empty
    assert len(cluster_dict) > 0
    assert 'feature' in cluster_df.columns
    assert 'cluster' in cluster_df.columns
    
    print(f"  Found {len(cluster_dict)} clusters")
    print("  ✓ Clustering test passed")
    return True


def test_engine_compute():
    """Test 7: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 7: Full Engine Computation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine(cluster_threshold=0.40)
    
    result = engine.compute(df)
    
    assert result.redundancy_index >= 0
    assert result.redundancy_index <= 100
    assert not result.redundancy_table.empty
    assert not result.clusters.empty
    assert not result.cluster_representatives.empty
    assert result.summary_statistics['total_features'] > 0
    
    print(f"  Redundancy index: {result.redundancy_index:.2f}/100")
    print(f"  Total features: {result.summary_statistics['total_features']}")
    print(f"  Unique clusters: {result.summary_statistics['unique_clusters']}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 8: Statistics engine."""
    print("\n" + "=" * 60)
    print("TEST 8: Statistics Engine")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine(cluster_threshold=0.40)

    print(f"  Redundancy index: {stats.redundancy_index:.2f}/100")
    print("  ✓ Statistics test passed")
    return True
