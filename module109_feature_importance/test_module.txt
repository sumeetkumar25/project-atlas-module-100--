def create_test_data(n=1000, seed=42):
    """Create synthetic test data with features and target."""
    np.random.seed(seed)

    state = np.random.randint(0, 16, n)
    persistence = np.random.randn(n) * 0.5 + state * 0.1
    volatility = np.random.randn(n) * 0.8 + np.sin(state) * 0.05
    noise = np.random.randn(n) * 1.0

    target_prob = 1 / (1 + np.exp(-(
        0.3 * (state - 7.5) / 7.5 +
        0.2 * persistence +
        0.1 * volatility +
        0.01 * noise
    )))
    target = (np.random.random(n) < target_prob).astype(int)

    return pd.DataFrame({
        'state': state,
        'persistence': persistence,
        'volatility': volatility,
        'noise': noise,
        'future_breakout': target
    })

def test_engine_correlation_filtering():
    """Test 3: Correlation filtering (backward compatible)."""
    print("\n" + "=" * 60)
    print("TEST 3: Correlation Filtering")
    print("=" * 60)

    engine = FeatureImportanceEngine(correlation_threshold=0.9)

    X = pd.DataFrame({
        'feature1': np.random.randn(100),
        'feature2': np.random.randn(100) * 0.5,
        'feature3': np.random.randn(100) * 0.5,
    })
    X['feature1_corr'] = X['feature1'] * 0.98 + np.random.randn(100) * 0.02
    X['feature1_high_corr'] = X['feature1'] * 0.99 + np.random.randn(100) * 0.01

    X_filtered, selected = engine.remove_correlated_features(X, threshold=0.9)

    assert len(selected) < len(X.columns)
    assert 'feature1' in selected or 'feature1_corr' in selected or 'feature1_high_corr' in selected

    print(f"  Generic mode: Original {len(X.columns)} → Selected {len(selected)} features")
    print("  ✓ Correlation filtering test passed")
    return True

def test_engine_permutation_importance():
    """Test 7: Permutation importance (Ensemble + CV)."""
    print("\n" + "=" * 60)
    print("TEST 7: Permutation Importance (Ensemble + CV)")
    print("=" * 60)

    engine = FeatureImportanceEngine(n_permutations=5, n_folds=3, include_state=False)
    df = create_test_data(n=300)
    X, y, features = engine.prepare_data(df)

    perm_scores = engine.compute_permutation_importance(X, y)

    assert len(perm_scores) == len(features)
    assert not perm_scores.isna().any()
    assert np.isfinite(perm_scores).all()

    # Signal feature (persistence) should outperform pure noise
    assert perm_scores["persistence"] > perm_scores["noise"]

    # Test stability across runs with fixed seed
    engine2 = FeatureImportanceEngine(n_permutations=5, n_folds=3, random_state=42, include_state=False)
    perm_scores2 = engine2.compute_permutation_importance(X, y)
    assert np.allclose(perm_scores.values, perm_scores2.values, atol=0.01)

    print(f"  Permutation importance (persistence): {perm_scores['persistence']:.4f}")
    print(f"  Permutation importance (noise): {perm_scores['noise']:.4f}")
    print("  ✓ Permutation importance test passed")
    return True
