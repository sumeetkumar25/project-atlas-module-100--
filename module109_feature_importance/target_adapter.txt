"""
Module 109 - Target Adapter

Connects Module 108 breakout targets to Module 109's target column.
"""

import pandas as pd


def attach_breakout_target(
    df: pd.DataFrame,
    breakout_targets: pd.DataFrame,
    horizon: int,
    target_column: str = "future_breakout",
) -> pd.DataFrame:
    """
    Attach a Module 108 breakout target to the Module 109 dataset.
    
    This establishes the dependency chain:
        Module 108 features → breakout_{horizon} → target_adapter → future_breakout → Module 109 target
    
    The original breakout_{horizon} column remains excluded from predictor selection
    via the engine's detect_features() and prepare_data() protections.
    
    Args:
        df: Original DataFrame (with features)
        breakout_targets: DataFrame from Module 108 with breakout_* columns
        horizon: Which horizon to use (e.g., 5 for breakout_5)
        target_column: Name to give the target column in the output (default: "future_breakout")
    
    Returns:
        DataFrame with the target column added
    
    Raises:
        ValueError: If the specified breakout target is not found or if data alignment fails
    """
    target_name = f"breakout_{horizon}"
    
    if target_name not in breakout_targets.columns:
        raise ValueError(
            f"Breakout target '{target_name}' not found. "
            f"Available breakout targets: "
            f"{[
                c for c in breakout_targets.columns
                if str(c).startswith('breakout_')
            ]}"
        )
    
    if len(df) != len(breakout_targets):
        raise ValueError(
            "Data alignment failure: source DataFrame and Module 108 "
            f"targets have different row counts "
            f"({len(df)} != {len(breakout_targets)})"
        )
    
    result = df.copy()
    
    target = breakout_targets[target_name].copy()
    
    if len(target) != len(result):
        raise ValueError(
            f"Target alignment failure for '{target_name}'"
        )
    
    result[target_column] = target.to_numpy()
    
    return result
