# atlas/modules/module119_feature_quality/report.py
"""
Module 119 - Feature Quality Report
"""

from datetime import datetime
import pandas as pd


class FeatureQualityReport:
    """Report generator for feature quality results."""
    
    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        """Generate a formatted report."""
        lines = []
        
        lines.append("=" * 55)
        lines.append("MODULE 119")
        lines.append("FEATURE QUALITY")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")
        
        # Overview
        lines.append("\n" + "-" * 45)
        lines.append("OVERVIEW")
        lines.append("-" * 45)
        lines.append(f"  Total Features: {statistics.total_features}")
        lines.append(f"  PASS: {statistics.passed_count}")
        lines.append(f"  WARNING: {statistics.warning_count}")
        lines.append(f"  FAIL: {statistics.failed_count}")
        
        # Percentages
        lines.append("\n" + "-" * 45)
        lines.append("PERCENTAGES")
        lines.append("-" * 45)
        lines.append(f"  PASS: {statistics.passed_percentage:.1f}%")
        lines.append(f"  WARNING: {statistics.warning_percentage:.1f}%")
        lines.append(f"  FAIL: {statistics.failed_percentage:.1f}%")
        
        # Quality Scores
        lines.append("\n" + "-" * 45)
        lines.append("QUALITY SCORES")
        lines.append("-" * 45)
        lines.append(f"  Average: {statistics.average_quality:.3f}")
        lines.append(f"  Median: {statistics.median_quality:.3f}")
        lines.append(f"  Best: {statistics.best_score:.3f}")
        lines.append(f"  Worst: {statistics.worst_score:.3f}")
        
        # Best and Worst Features
        if statistics.best_feature:
            lines.append("\n" + "-" * 45)
            lines.append("BEST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.best_feature}: {statistics.best_score:.3f}")
        
        if statistics.worst_feature:
            lines.append("\n" + "-" * 45)
            lines.append("WORST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.worst_feature}: {statistics.worst_score:.3f}")
        
        # Metrics
        lines.append("\n" + "-" * 45)
        lines.append("AVERAGE METRICS")
        lines.append("-" * 45)
        lines.append(f"  Missing: {statistics.average_missing:.1f}%")
        lines.append(f"  Skewness: {statistics.average_skew:.3f}")
        lines.append(f"  Outliers: {statistics.average_outliers:.1f}%")
        
        # Interpretation
        lines.append("\n" + "-" * 45)
        lines.append("INTERPRETATION")
        lines.append("-" * 45)
        
        if statistics.passed_percentage > 70:
            lines.append("  ✅ Most features passed quality checks")
        elif statistics.passed_percentage > 50:
            lines.append("  ✓ Moderate number of features passed")
        else:
            lines.append("  ⚠ Many features failed quality checks")
        
        if statistics.average_missing > 10:
            lines.append(f"  ⚠ High missing values: {statistics.average_missing:.1f}% average")
        
        # Validation
        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        
        # Footer
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)
        
        return "\n".join(lines)
    
    @staticmethod
    def generate_quality_table(quality_table: pd.DataFrame, top_n: int = 10) -> str:
        """Generate a quality table report."""
        lines = []
        lines.append("\n" + "=" * 80)
        lines.append(f"TOP {top_n} FEATURES BY QUALITY")
        lines.append("=" * 80)
        
        top_features = quality_table.head(top_n)
        
        lines.append(
            f"\n{'Feature':<20} {'Score':<10} {'Decision':<12} {'Missing%':<10} {'Skew':<10} {'Outliers%':<10}"
        )
        lines.append("-" * 80)
        
        for _, row in top_features.iterrows():
            lines.append(
                f"{row['feature'][:19]:<20} "
                f"{row['quality_score']:<10.3f} "
                f"{row['decision']:<12} "
                f"{row['missing_pct']:<10.1f} "
                f"{row['skewness']:<10.3f} "
                f"{row['outlier_pct']:<10.1f}"
            )
        
        lines.append("\n" + "=" * 80)
        return "\n".join(lines)
    
    @staticmethod
    def save(report: str, filepath: str):
        """Save report to file."""
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
