def create_test_features(n=100, seed=42):
    """Create synthetic test features."""
    np.random.seed(seed)
    
    # Create timestamps
    timestamps = pd.date_range('2020-01-01', periods=n, freq='1h')
    
    # Feature 1: Trend score
    trend_df = pd.DataFrame({
        'timestamp': timestamps,
        'trend_score': np.random.randn(n) * 10 + 50
    })
    
    # Feature 2: Compression score (with some missing)
    compression = np.random.randn(n) * 10 + 50
    compression[:10] = np.nan  # Introduce some missing values
    compression_df = pd.DataFrame({
        'timestamp': timestamps[5:],  # Different timestamp alignment
        'compression_score': compression[5:]
    })
    
    # Feature 3: Momentum score
    momentum_df = pd.DataFrame({
        'timestamp': timestamps,
        'momentum_score': np.random.randn(n) * 5 + 50
    })
    
    # Feature 4: Liquidity score (constant feature)
    liquidity_df = pd.DataFrame({
        'timestamp': timestamps,
        'liquidity_score': np.ones(n) * 5
    })
    
    # Feature 5: Volume score (duplicate timestamp test)
    volume_df = pd.DataFrame({
        'timestamp': timestamps[:90],
        'volume_score': np.random.randn(90) * 100 + 1000
    })
    
    return {
        'trend': trend_df,
        'compression': compression_df,
        'momentum': momentum_df,
        'liquidity': liquidity_df,
        'volume': volume_df
    }


def test_engine_feature_loading():
    """Test 1: Feature loading and merging."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Loading and Merging")
    print("=" * 60)
    
    feature_dict = create_test_features()
    engine = FeatureRepositoryEngine()
    
    master_df = engine.load_features(feature_dict)
    
    assert not master_df.empty
    assert 'timestamp' in master_df.columns
    assert 'trend_score' in master_df.columns
    assert 'compression_score' in master_df.columns
    assert 'momentum_score' in master_df.columns
    assert 'liquidity_score' in master_df.columns
    
    print(f"  Master shape: {master_df.shape}")
    print("  ✓ Feature loading test passed")
    return True


def test_engine_timestamp_alignment():
    """Test 2: Timestamp alignment."""
    print("\n" + "=" * 60)
    print("TEST 2: Timestamp Alignment")
    print("=" * 60)
    
    feature_dict = create_test_features()
    engine = FeatureRepositoryEngine()
    
    master_df = engine.load_features(feature_dict)
    master_df = engine.align_timestamps(master_df)
    
    # Check sorted
    assert master_df['timestamp'].is_monotonic_increasing
    
    # Check timestamps are datetime
    assert pd.api.types.is_datetime64_any_dtype(master_df['timestamp'])
    
    print("  ✓ Timestamp alignment test passed")
    return True


def test_engine_metadata():
    """Test 3: Metadata generation."""
    print("\n" + "=" * 60)
    print("TEST 3: Metadata Generation")
    print("=" * 60)
    
    feature_dict = create_test_features()
    engine = FeatureRepositoryEngine()
    
    master_df = engine.load_features(feature_dict)
    master_df = engine.align_timestamps(master_df)
    metadata = engine.build_metadata(master_df)
    
    assert not metadata.empty
    assert 'feature' in metadata.columns
    assert 'null_percentage' in metadata.columns
    assert 'unique_values' in metadata.columns
    
    # Should identify liquidity_score as constant
    liquidity_row = metadata[metadata['feature'] == 'liquidity_score']
    if not liquidity_row.empty:
        assert liquidity_row['unique_values'].iloc[0] == 1
    
    print(f"  Metadata shape: {metadata.shape}")
    print("  ✓ Metadata test passed")
    return True


def test_engine_statistics():
    """Test 4: Statistics computation."""
    print("\n" + "=" * 60)
    print("TEST 4: Statistics Computation")
    print("=" * 60)
    
    feature_dict = create_test_features()
    engine = FeatureRepositoryEngine(fill_method='ffill')
    
    master_df = engine.load_features(feature_dict)
    master_df = engine.align_timestamps(master_df)
    master_df = engine.handle_missing_values(master_df)
    metadata = engine.build_metadata(master_df)
    stats = engine.compute_statistics(master_df, metadata)
    
    assert stats['total_rows'] > 0
    assert stats['total_features'] > 0
    assert stats['memory_usage_mb'] > 0
    assert 0 <= stats['feature_density'] <= 1
    assert 0 <= stats['missing_percentage'] <= 100
    
    print(f"  Total rows: {stats['total_rows']}")
    print(f"  Total features: {stats['total_features']}")
    print(f"  Missing %: {stats['missing_percentage']:.2f}%")
    print("  ✓ Statistics test passed")
    return True


def test_engine_fill_methods():
    """Test 5: Fill methods."""
    print("\n" + "=" * 60)
    print("TEST 5: Fill Methods")
    print("=" * 60)
    
    feature_dict = create_test_features()
    
    # Test forward fill
    engine_ffill = FeatureRepositoryEngine(fill_method='ffill')
    master_ffill = engine_ffill.load_features(feature_dict)
    master_ffill = engine_ffill.align_timestamps(master_ffill)
    master_ffill = engine_ffill.handle_missing_values(master_ffill)
    
    # Test no fill
    engine_none = FeatureRepositoryEngine(fill_method='none')
    master_none = engine_none.load_features(feature_dict)
    master_none = engine_none.align_timestamps(master_none)
    master_none = engine_none.handle_missing_values(master_none)
    
    # Forward fill should have fewer NaN values
    assert master_ffill.isna().sum().sum() <= master_none.isna().sum().sum()
    
    print(f"  Ffill NaN count: {master_ffill.isna().sum().sum()}")
    print(f"  None NaN count: {master_none.isna().sum().sum()}")
    print("  ✓ Fill methods test passed")
    return True

def test_statistics():
    """Test 7: Statistics engine."""
