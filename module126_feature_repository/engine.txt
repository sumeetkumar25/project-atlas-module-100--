# atlas/modules/module126_feature_repository/engine.py
"""
Module 126 - Feature Repository Engine

Collects every engineered feature into one standardized dataframe.

Responsibilities:
  1. Automatically discover feature files from feature_store
  2. Load all feature DataFrames
  3. Align timestamps
  4. Merge all features
  5. Validate
  6. Remove duplicates
  7. Sort columns
  8. Output master feature matrix
  9. Save master matrix to data/master/
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Optional, Dict, Any, List, Tuple
from pathlib import Path
import warnings
warnings.filterwarnings('ignore')


@dataclass
class FeatureRepositoryResult:
    """Results from feature repository analysis."""
    master_feature_matrix: pd.DataFrame
    feature_metadata: pd.DataFrame
    summary_statistics: Dict[str, Any]


class FeatureRepositoryEngine:
    """Core engine for feature repository."""

    # Required columns for validation
    REQUIRED_COLS = {'timestamp'}

    # Columns to exclude from feature counting
    IGNORE_COLS = {'timestamp', 'date', 'time', 'index'}

    def __init__(
        self,
        timestamp_column: str = "timestamp",
        drop_duplicates: bool = True,
        fill_method: str = "ffill"
    ):
        """
        Initialize the feature repository engine.
        
        Args:
            timestamp_column: Name of the timestamp column
            drop_duplicates: Whether to drop duplicate timestamps
            fill_method: Method for filling missing values ('ffill', 'bfill', 'none')
        """
        self.timestamp_column = timestamp_column
        self.drop_duplicates = drop_duplicates
        self.fill_method = fill_method

    def discover_feature_files(self, feature_store: Path) -> List[Path]:
        """
        Automatically discover every feature file.
        
        Args:
            feature_store: Path to the feature store directory
            
        Returns:
            List of feature file paths
        """
        if not feature_store.exists():
            raise FileNotFoundError(f"Feature store not found: {feature_store}")
        
        # Search for parquet files
        files = sorted(feature_store.glob("*.parquet"))
        
        # Also search for CSV files (backward compatibility)
        csv_files = sorted(feature_store.glob("*.csv"))
        files.extend(csv_files)
        
        if not files:
            raise FileNotFoundError(
                f"No feature files found in {feature_store}. "
                f"Expected .parquet or .csv files."
            )
        
        return files

    def load_feature_store(self, feature_store: Path) -> Dict[str, pd.DataFrame]:
        """
        Load every discovered feature from the feature store.
        
        Args:
            feature_store: Path to the feature store directory
            
        Returns:
            Dictionary mapping feature name to DataFrame
        """
        files = self.discover_feature_files(feature_store)
        
        feature_dict = {}
        
        for file in files:
            # Determine file type and load
            if file.suffix == '.parquet':
                df = pd.read_parquet(file)
            elif file.suffix == '.csv':
                df = pd.read_csv(file)
                # Convert timestamp column to datetime if needed
                if self.timestamp_column in df.columns:
                    df[self.timestamp_column] = pd.to_datetime(df[self.timestamp_column])
            else:
                continue
            
            feature_dict[file.stem] = df
        
        print(f"  Loaded {len(feature_dict)} feature files from {feature_store}")
        return feature_dict

    def load_features(self, feature_dict: Dict[str, pd.DataFrame]) -> pd.DataFrame:
        """
        Load and merge multiple feature DataFrames.
        
        Args:
            feature_dict: Dictionary mapping feature name to DataFrame
            
        Returns:
            Merged master feature matrix
        """
        if not feature_dict:
            raise ValueError("Feature dictionary is empty")
        
        # Start with first DataFrame
        first_key = list(feature_dict.keys())[0]
        master_df = feature_dict[first_key].copy()
        
        # Merge remaining DataFrames
        for key, df in list(feature_dict.items())[1:]:
            # Ensure timestamp column exists
            if self.timestamp_column not in df.columns:
                raise ValueError(f"DataFrame '{key}' missing timestamp column")
            
            # Convert timestamp to datetime if needed
            if not pd.api.types.is_datetime64_any_dtype(df[self.timestamp_column]):
                df[self.timestamp_column] = pd.to_datetime(df[self.timestamp_column])
            
            # Merge on timestamp
            master_df = pd.merge(
                master_df,
                df,
                on=self.timestamp_column,
                how='outer'
            )
        
        return master_df

    def align_timestamps(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Align timestamps and ensure chronological order.
        """
        # Ensure timestamp is datetime
        if not pd.api.types.is_datetime64_any_dtype(df[self.timestamp_column]):
            df[self.timestamp_column] = pd.to_datetime(df[self.timestamp_column])
        
        # Sort by timestamp
        df = df.sort_values(self.timestamp_column).reset_index(drop=True)
        
        # Remove duplicate timestamps if requested
        if self.drop_duplicates:
            df = df.drop_duplicates(subset=[self.timestamp_column], keep='first')
        
        return df

    def handle_missing_values(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Handle missing values in the feature matrix.
        Compatible with Pandas 2.x and 3.x.
        """
        if self.fill_method == "ffill":
            df = df.ffill()
            df = df.bfill()
        
        elif self.fill_method == "bfill":
            df = df.bfill()
            df = df.ffill()
        
        elif self.fill_method == "none":
            pass
        
        else:
            raise ValueError(
                f"Unknown fill method: {self.fill_method}"
            )
        
        return df

    def get_feature_columns(self, df: pd.DataFrame) -> List[str]:
        """
        Identify feature columns (exclude timestamp and metadata).
        """
        features = [
            c for c in df.columns
            if c not in self.IGNORE_COLS
            and c != self.timestamp_column
        ]
        return features

    def build_metadata(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Build feature metadata table.
        """
        features = self.get_feature_columns(df)
        
        metadata_rows = []
        for feature in features:
            metadata_rows.append({
                'feature': feature,
                'dtype': str(df[feature].dtype),
                'non_null': df[feature].count(),
                'null_count': df[feature].isna().sum(),
                'null_percentage': df[feature].isna().mean() * 100,
                'unique_values': df[feature].nunique(),
                'min_value': df[feature].min() if pd.api.types.is_numeric_dtype(df[feature]) else None,
                'max_value': df[feature].max() if pd.api.types.is_numeric_dtype(df[feature]) else None,
                'mean_value': df[feature].mean() if pd.api.types.is_numeric_dtype(df[feature]) else None,
                'std_value': df[feature].std() if pd.api.types.is_numeric_dtype(df[feature]) else None
            })
        
        return pd.DataFrame(metadata_rows)

    def compute_statistics(
        self,
        df: pd.DataFrame,
        metadata: pd.DataFrame
    ) -> Dict[str, Any]:
        """
        Compute summary statistics.
        """
        features = self.get_feature_columns(df)
        
        # Memory usage
        memory_usage = df.memory_usage(deep=True).sum() / (1024 * 1024)  # MB
        
        # Feature density
        feature_density = 1 - (metadata['null_percentage'].mean() / 100)
        
        # Time range
        time_range = None
        if self.timestamp_column in df.columns:
            time_range = {
                'start': df[self.timestamp_column].min(),
                'end': df[self.timestamp_column].max(),
                'duration_days': (df[self.timestamp_column].max() - df[self.timestamp_column].min()).days
            }
        
        stats = {
            'total_rows': len(df),
            'total_features': len(features),
            'total_columns': len(df.columns),
            'memory_usage_mb': memory_usage,
            'feature_density': feature_density,
            'missing_percentage': metadata['null_percentage'].mean(),
            'duplicate_timestamps': df[self.timestamp_column].duplicated().sum() if self.timestamp_column in df.columns else 0,
            'constant_features': sum(metadata['unique_values'] == 1),
            'time_range': time_range,
            'fill_method': self.fill_method
        }
        
        return stats

    def save_master_matrix(self, df: pd.DataFrame, metadata: pd.DataFrame) -> None:
        """
        Save the master feature matrix and metadata to the data/master directory.
        """
        master_dir = Path("data/master")
        master_dir.mkdir(parents=True, exist_ok=True)
        
        # Save master matrix
        master_path = master_dir / "master_feature_matrix.parquet"
        df.to_parquet(master_path, index=False)
        print(f"  Saved master matrix to {master_path}")
        
        # Save metadata
        metadata_path = master_dir / "feature_metadata.parquet"
        metadata.to_parquet(metadata_path, index=False)
        print(f"  Saved feature metadata to {metadata_path}")

    def compute(
        self,
        feature_store: Optional[Path] = None,
        feature_dict: Optional[Dict[str, pd.DataFrame]] = None
    ) -> FeatureRepositoryResult:
        """
        Main computation method.
        
        Args:
            feature_store: Path to the feature store directory (auto-discovers files)
            feature_dict: Dictionary of feature DataFrames (legacy support)
            
        Returns:
            FeatureRepositoryResult with master matrix and metadata
        """
        print("\n  Building Feature Repository...")
        
        # Load features either from feature store or from dictionary
        if feature_store is not None:
            print(f"  Loading feature store from: {feature_store}")
            feature_dict = self.load_feature_store(Path(feature_store))
        elif feature_dict is not None:
            print(f"  Loading from feature dictionary with {len(feature_dict)} features...")
        else:
            raise ValueError("Either feature_store or feature_dict must be provided")
        
        print(f"  Processing {len(feature_dict)} feature DataFrames...")
        
        # Load and merge features
        master_df = self.load_features(feature_dict)
        print(f"  Merged shape: {master_df.shape}")
        
        # Align timestamps
        master_df = self.align_timestamps(master_df)
        print(f"  Aligned shape: {master_df.shape}")
        
        # Handle missing values
        master_df = self.handle_missing_values(master_df)
        
        # Build metadata
        metadata = self.build_metadata(master_df)
        
        # Compute statistics
        stats = self.compute_statistics(master_df, metadata)
        
        print(f"\n  Repository Summary:")
        print(f"    Rows: {stats['total_rows']:,}")
        print(f"    Features: {stats['total_features']}")
        print(f"    Missing: {stats['missing_percentage']:.2f}%")
        print(f"    Memory: {stats['memory_usage_mb']:.2f} MB")
        
        # Save master matrix to data/master
        self.save_master_matrix(master_df, metadata)
        
        return FeatureRepositoryResult(
            master_feature_matrix=master_df,
            feature_metadata=metadata,
            summary_statistics=stats
        )
