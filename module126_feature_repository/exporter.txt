"""
Module 126 - Feature Repository Exporter
"""

import pandas as pd
from pathlib import Path


class FeatureRepositoryExporter:
    """Exporter for feature repository results."""
    
    @staticmethod
    def export_master_matrix(df: pd.DataFrame, filepath: str):
        """Export master feature matrix to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_metadata(df: pd.DataFrame, filepath: str):
        """Export feature metadata to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_summary(summary: dict, filepath: str):
        """Export summary to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame([summary])
        df.to_csv(path, index=False)
