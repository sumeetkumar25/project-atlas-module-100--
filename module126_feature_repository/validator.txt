"""
Module 126 - Feature Repository Validator
"""

import numpy as np
import pandas as pd
from .engine import FeatureRepositoryResult
from .statistics import FeatureRepositoryStatistics


class FeatureRepositoryValidator:
    """Validator for feature repository results."""
    
    def validate(
        self,
        result: FeatureRepositoryResult,
        statistics: FeatureRepositoryStatistics
    ) -> bool:
        """Validate results and statistics."""
        issues = []
        warnings = []
        
        # Check master matrix
        if result.master_feature_matrix.empty:
            issues.append("Master feature matrix is empty")
        else:
            # Check timestamp column
            if 'timestamp' not in result.master_feature_matrix.columns:
                issues.append("Missing timestamp column")
            else:
                # Check duplicate timestamps
                dup_count = result.master_feature_matrix['timestamp'].duplicated().sum()
                if dup_count > 0:
                    warnings.append(f"Found {dup_count} duplicate timestamps")
            
            # Check NaN ratio
            nan_pct = result.master_feature_matrix.isna().mean().mean() * 100
            if nan_pct > 10:
                warnings.append(f"High NaN ratio: {nan_pct:.1f}%")
            
            # Check constant columns
            constant_cols = []
            for col in result.master_feature_matrix.columns:
                if col != 'timestamp' and result.master_feature_matrix[col].nunique() == 1:
                    constant_cols.append(col)
            if constant_cols:
                warnings.append(f"Found {len(constant_cols)} constant columns: {constant_cols[:5]}...")
        
        # Check metadata
        if result.feature_metadata.empty:
            issues.append("Feature metadata is empty")
        else:
            # Check required columns
            required_cols = ['feature', 'dtype', 'null_count', 'null_percentage', 'unique_values']
            missing = set(required_cols) - set(result.feature_metadata.columns)
            if missing:
                issues.append(f"Missing columns in metadata: {missing}")
            
            # Check feature names are unique
            if result.feature_metadata['feature'].duplicated().any():
                issues.append("Duplicate feature names found in metadata")
        
        # Check statistics
        if statistics.total_rows <= 0:
            issues.append(f"Invalid total rows: {statistics.total_rows}")
        
        if statistics.total_features <= 0:
            issues.append(f"Invalid total features: {statistics.total_features}")
        
        if not (0 <= statistics.feature_density <= 1):
            issues.append(f"Feature density outside [0,1]: {statistics.feature_density}")
        
        if not (0 <= statistics.missing_percentage <= 100):
            issues.append(f"Missing percentage outside [0,100]: {statistics.missing_percentage}")
        
        # Check statistics match result
        if statistics.total_rows != len(result.master_feature_matrix):
            issues.append(
                f"Total rows mismatch: stats={statistics.total_rows}, "
                f"result={len(result.master_feature_matrix)}"
            )
        
        if statistics.total_features != len(result.feature_metadata):
            issues.append(
                f"Total features mismatch: stats={statistics.total_features}, "
                f"result={len(result.feature_metadata)}"
            )
        
        # Warnings (non-fatal)
        if statistics.missing_percentage > 5:
            warnings.append(f"High missing percentage: {statistics.missing_percentage:.1f}%")
        
        if statistics.constant_features > 0:
            warnings.append(f"Found {statistics.constant_features} constant features")
        
        if warnings:
            print("\n  WARNINGS:")
            for warning in warnings:
                print(f"    ⚠ {warning}")
        
        if issues:
            raise ValueError(
                f"Feature Repository validation failed with {len(issues)} issue(s):\n" +
                "\n".join(f"  - {issue}" for issue in issues)
            )
        
        return True


def run_validation(result, statistics):
    """Run validation with user-friendly output."""
    validator = FeatureRepositoryValidator()
    
    print("\n" + "=" * 60)
    print("FEATURE REPOSITORY VALIDATION")
    print("=" * 60)
    
    try:
        validator.validate(result, statistics)
        print("\n  ✓ All validation checks passed")
        print("=" * 60)
        return True
    except ValueError as e:
        print(e)
        print("=" * 60)
        return False
