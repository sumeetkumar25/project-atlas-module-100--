"""
Module 101
Feature Validation Engine

Evaluates whether a feature predicts future BTC returns.
Accepts a feature column and close prices as input.

Pipeline: feature + close → future returns → IC → buckets → statistics → validation → report → export
"""

from typing import Optional, Dict, Any

import pandas as pd

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureValidationEngine
from .statistics import FeatureValidationStatisticsEngine
from .validator import run_validation
from .report import FeatureValidationReport


class FeatureValidationResearch(BaseResearchModule):

    MODULE_OUTPUT = "module101"
    METADATA = FeatureValidationStatisticsEngine.METADATA

    def __init__(self, registry: Optional[AtlasRegistry] = None, horizon: int = 10):

        super().__init__(
            module_name="Feature Validation",
            module_version=self.METADATA["version"],
        )

        self.registry = registry or AtlasRegistry()
        self.engine = FeatureValidationEngine(horizon=horizon, buckets=10)
        self.stats_engine = FeatureValidationStatisticsEngine()
        self.result_fv = None
        self.statistics = None

    def _register_pipeline_steps(self):
        self.pipeline_steps = [
            ("feature_validation", self._compute_validation),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]

    def _compute_validation(self) -> Dict[str, Any]:
        self.logger.info(f"Step 1: Computing feature validation (horizon={self.engine.horizon})")

        close = self.df["close"].values
        
        # Use trend_score from Module 100
        feature_column = "trend_score"
        
        if feature_column not in self.df.columns:
            raise RuntimeError(
                f"Expected '{feature_column}' from Module 100. "
                f"Available columns: {list(self.df.columns)}"
            )
        
        feature = self.df[feature_column].values

        self.result_fv = self.engine.compute(close, feature, series_type="feature_validation")

        self.logger.metric("Horizon", f"{self.result_fv.horizon}")
        self.logger.metric("IC", f"{self.result_fv.information_coefficient:.6f}")
        self.logger.metric("Spearman", f"{self.result_fv.spearman_correlation:.6f}")
        self.logger.metric("Pearson", f"{self.result_fv.pearson_correlation:.6f}")
        self.logger.metric("Sample size", f"{self.result_fv.sample_size:,}")
        self.logger.success("Feature validation computed")

        return {
            "information_coefficient": self.result_fv.information_coefficient,
            "spearman": self.result_fv.spearman_correlation,
            "pearson": self.result_fv.pearson_correlation,
            "sample_size": self.result_fv.sample_size,
        }

    def _compute_statistics(self) -> Dict[str, Any]:
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_fv)
        self.logger.metric("Regime", self.statistics.regime)
        self.logger.metric("Monotonicity", f"{self.statistics.bucket_monotonicity:.1f}%")
        self.logger.success("Statistics computed")
        return self.stats_engine.summary_dict(self.statistics)

    def _validate(self) -> Dict[str, Any]:
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_fv, self.statistics)
        self.logger.success("Validation complete")
        return {"validation_passed": passed}

    def _export_results(self) -> Dict[str, Any]:
        self.logger.info("Step 4: Export")

        self.result.artifacts["feature_validation_statistics"] = (
            self.stats_engine.summary_dataframe(self.statistics)
        )

        bucket_df = pd.DataFrame([
            {
                "bucket": v["bucket"],
                "count": v["count"],
                "mean_return": v["mean_return"],
                "median_return": v["median_return"],
                "win_rate": v["win_rate"],
            }
            for v in self.result_fv.bucket_statistics.values()
        ])
        self.result.artifacts["bucket_statistics"] = bucket_df

        fv_df = pd.DataFrame({
            "feature_values": self.result_fv.feature_values,
            "future_returns": self.result_fv.future_returns,
        })
        self.result.artifacts["feature_validation_data"] = fv_df

        report = FeatureValidationReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_validation_report"] = report

        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
        }

    def compute(self) -> Dict[str, Any]:
        self.logger.section("Feature Validation Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 101 complete")
        return results

    def run(self, filepath: str, output_dir=None):
        return super().run(filepath, output_dir)

    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        return super().run_dataframe(df, dataset_id, output_dir)
