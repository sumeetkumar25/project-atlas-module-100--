"""
Module 116 - Feature Interaction Exporter
"""

import pandas as pd
from pathlib import Path


class FeatureInteractionExporter:
    """Exporter for feature interaction results."""
    
    @staticmethod
    def export_interaction_table(df: pd.DataFrame, filepath: str):
        """Export interaction table to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_interaction_matrix(df: pd.DataFrame, filepath: str):
        """Export interaction matrix to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path)
    
    @staticmethod
    def export_top_interactions(df: pd.DataFrame, filepath: str):
        """Export top interactions to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_summary(summary: dict, filepath: str):
        """Export summary to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame([summary])
        df.to_csv(path, index=False)
