def create_test_data(n=500, seed=42):
    """Create synthetic test data with feature interactions."""
    np.random.seed(seed)
    
    # Create features with known interactions
    x1 = np.random.randn(n)
    x2 = np.random.randn(n)
    x3 = np.random.randn(n) * 0.3 + x1 * 0.3
    x4 = np.random.randn(n)  # Pure noise
    x5 = np.random.randn(n) * 0.5 + x2 * 0.5
    
    # Target with interactions: x1*x2 and x1*x3
    y_prob = 1 / (1 + np.exp(-(
        0.3 * x1 * x2 + 
        0.2 * x1 * x3 +
        0.1 * x2 +
        0.1 * x1
    )))
    y = (np.random.random(n) < y_prob).astype(int)
    
    df = pd.DataFrame({
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'feature5': x5,
        'future_breakout': y
    })
    
    return df

    print("TEST 1: Feature Detection")

    return True

def test_engine_interaction_gain():
    """Test 5: Interaction gain and classification."""
    print("\n" + "=" * 60)
    print("TEST 5: Interaction Gain and Classification")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureInteractionEngine(min_samples=10)
    y = df['future_breakout'].values
    
    # Known interacting pair
    score_a = engine.compute_feature_score(df, 'feature1', y)
    score_b = engine.compute_feature_score(df, 'feature2', y)
    combined = engine.compute_combined_score(df, 'feature1', 'feature2', y)
    
    max_individual = max(score_a, score_b)
    gain = combined - max_individual
    synergy = np.clip(gain / (max_individual + 1e-12), 0, 1)
    
    # Test gain and synergy bounds
    assert -1.0 <= gain <= 1.0
    assert 0.0 <= synergy <= 1.0
    
    # Test classification
    interaction = engine.classify_interaction(synergy, gain)
    assert interaction in {
        "Strong Synergy",
        "Moderate Synergy",
        "Independent",
        "Conflict"
    }
    
    print(f"  Gain: {gain:.4f}")
    print(f"  Synergy: {synergy:.4f}")
    print(f"  Classification: {interaction}")
    print("  ✓ Interaction gain and classification test passed")
    return True


def test_engine_interaction_type():
    """Test 6: Interaction classification boundaries."""
    print("\n" + "=" * 60)
    print("TEST 6: Interaction Classification Boundaries")
    print("=" * 60)
    
    engine = FeatureInteractionEngine()
    
    # Strong Synergy threshold (0.25)
    # Just below threshold -> Moderate Synergy
    assert engine.classify_interaction(0.2499, 0.15) == "Moderate Synergy"
    # At threshold -> Moderate Synergy (since > 0.25, not >=)
    assert engine.classify_interaction(0.2500, 0.15) == "Moderate Synergy"
    # Just above threshold -> Strong Synergy
    assert engine.classify_interaction(0.2501, 0.15) == "Strong Synergy"
    
    # Moderate Synergy threshold (0.10)
    # Just below threshold -> Independent
    assert engine.classify_interaction(0.0999, 0.05) == "Independent"
    # At threshold -> Independent (since > 0.10, not >=)
    assert engine.classify_interaction(0.1000, 0.05) == "Independent"
    # Just above threshold -> Moderate Synergy
    assert engine.classify_interaction(0.1001, 0.05) == "Moderate Synergy"
    
    # Conflict threshold (-0.05)
    # Just above threshold -> Independent
    assert engine.classify_interaction(0.01, -0.0499) == "Independent"
    # At threshold -> Independent (since < -0.05, not <=)
    assert engine.classify_interaction(0.01, -0.0500) == "Independent"
    # Just below threshold -> Conflict
    assert engine.classify_interaction(0.01, -0.0501) == "Conflict"
    
    # Additional sanity checks
    assert engine.classify_interaction(0.30, 0.20) == "Strong Synergy"
    assert engine.classify_interaction(0.15, 0.10) == "Moderate Synergy"
    assert engine.classify_interaction(0.05, 0.03) == "Independent"
    assert engine.classify_interaction(0.01, -0.10) == "Conflict"
    
    print("  ✓ Interaction classification boundaries test passed")
    return True

def test_performance_optimization():
    """Test 11: Performance optimization - cached individual scores."""
    print("\n" + "=" * 60)
    print("TEST 11: Performance Optimization")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureInteractionEngine(min_samples=10)
    y = df['future_breakout'].values
    
    # Get features
    features = engine.get_feature_columns(df)
    
    # Compute interactions with caching
    pairs = engine.generate_pairs(features)
    
    # Cache individual feature scores (this is the optimization)
    feature_scores = {
        feature: engine.compute_feature_score(df, feature, y)
        for feature in features
    }
    
    # Verify all features have scores
    assert len(feature_scores) == len(features)
    for feature in features:
        assert feature in feature_scores
        assert 0 <= feature_scores[feature] <= 1
    
    # Verify cached scores match direct computation
    for feature in features[:3]:  # Spot check first 3
        direct_score = engine.compute_feature_score(df, feature, y)
        assert np.isclose(feature_scores[feature], direct_score, atol=1e-6)
    
    print(f"  Cached scores for {len(feature_scores)} features")
    print("  ✓ Performance optimization test passed")
    return True
