"""
Module 110 - Feature Interaction Statistics
"""

from dataclasses import dataclass, asdict
import pandas as pd
from .engine import FeatureInteractionResult


@dataclass
class FeatureInteractionStatistics:
    """High-level statistics from feature interaction analysis."""

    total_pairs: int
    top_pair: str
    top_score: float
    top_stability: float
    interaction_entropy: float
    method_agreement: float
    agreement_level: str
    top_10_pairs: list
    methods_used: list
    n_features: int
    n_bootstrap: int
    n_folds: int


class FeatureInteractionStatisticsEngine:
    """Engine for generating high-level statistics."""
    
    METADATA = {
        "module": 110,
        "name": "Feature Interaction",
        "version": "2.0.0",
        "description": "Research-grade analysis of pairwise feature interactions for future breakout classification"
    }
    
    def analyze(
        self,
        result: FeatureInteractionResult
    ) -> FeatureInteractionStatistics:

        return FeatureInteractionStatistics(
            total_pairs=result.statistics["total_pairs"],
            top_pair=result.statistics["top_pair"],
            top_score=result.statistics["top_score"],
            top_stability=result.statistics["top_stability"],
            interaction_entropy=result.statistics[
                "interaction_entropy"
            ],
            method_agreement=result.statistics[
                "method_agreement"
            ],
            agreement_level=result.statistics[
                "agreement_level"
            ],
            top_10_pairs=result.statistics[
                "top_10_pairs"
            ],
            methods_used=result.statistics[
                "methods_used"
            ],
            n_features=result.statistics[
                "n_features"
            ],
            n_bootstrap=result.statistics[
                "n_bootstrap"
            ],
            n_folds=result.statistics[
                "n_folds"
            ],
        )
    
    def summary_dataframe(self, stats: FeatureInteractionStatistics) -> pd.DataFrame:
        """Convert statistics to DataFrame."""
        return pd.DataFrame([asdict(stats)])
    
    def summary_dict(self, stats: FeatureInteractionStatistics) -> dict:
        """Convert statistics to dictionary."""
        return asdict(stats)
