"""
Module 124 - Feature Redundancy Statistics
"""

from dataclasses import dataclass, asdict
import pandas as pd
from .engine import FeatureRedundancyResult


@dataclass
class FeatureRedundancyStatistics:
    """High-level statistics from feature redundancy analysis."""
    total_features: int
    kept_count: int
    removed_count: int
    kept_percentage: float
    removed_percentage: float
    average_redundancy: float
    median_redundancy: float
    min_redundancy: float
    max_redundancy: float
    largest_cluster: int
    smallest_cluster: int
    average_cluster_size: float
    best_feature: str
    worst_feature: str
    best_score: float
    worst_score: float
    total_clusters: int


class FeatureRedundancyStatisticsEngine:
    """Engine for generating high-level statistics."""
    
    METADATA = {
        "module": 124,
        "name": "Feature Redundancy Analysis",
        "version": "1.0.0",
        "description": "Determines which features carry the same information"
    }
    
    def analyze(self, result: FeatureRedundancyResult) -> FeatureRedundancyStatistics:
        """Analyze results and generate high-level statistics."""
        return FeatureRedundancyStatistics(
            total_features=result.summary_statistics['total_features'],
            kept_count=result.summary_statistics['kept_count'],
            removed_count=result.summary_statistics['removed_count'],
            kept_percentage=result.summary_statistics['kept_percentage'],
            removed_percentage=result.summary_statistics['removed_percentage'],
            average_redundancy=result.summary_statistics['average_redundancy'],
            median_redundancy=result.summary_statistics['median_redundancy'],
            min_redundancy=result.summary_statistics['min_redundancy'],
            max_redundancy=result.summary_statistics['max_redundancy'],
            largest_cluster=result.summary_statistics['largest_cluster'],
            smallest_cluster=result.summary_statistics['smallest_cluster'],
            average_cluster_size=result.summary_statistics['average_cluster_size'],
            best_feature=result.summary_statistics['best_feature'],
            worst_feature=result.summary_statistics['worst_feature'],
            best_score=result.summary_statistics['best_score'],
            worst_score=result.summary_statistics['worst_score'],
            total_clusters=result.summary_statistics['total_clusters']
        )
    
    def summary_dataframe(self, stats: FeatureRedundancyStatistics) -> pd.DataFrame:
        """Convert statistics to DataFrame."""
        return pd.DataFrame([asdict(stats)])
    
    def summary_dict(self, stats: FeatureRedundancyStatistics) -> dict:
        """Convert statistics to dictionary."""
        return asdict(stats)
