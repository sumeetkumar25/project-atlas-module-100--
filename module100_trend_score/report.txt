"""
Module 100 - Trend Score Report
"""

from datetime import datetime


class TrendScoreReport:

    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        lines = []

        lines.append("=" * 55)
        lines.append("MODULE 100")
        lines.append("COMPOSITE TREND SCORE")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")

        lines.append("\n" + "-" * 45)
        lines.append("COMPOSITE SCORE")
        lines.append("-" * 45)
        
        # Use dictionary access instead of attribute access
        if isinstance(statistics, dict):
            trend_score = statistics.get("trend_score", 0.0)
            median_score = statistics.get("median_score", 0.0)
            std_score = statistics.get("std_score", 0.0)
            min_score = statistics.get("min_score", 0.0)
            max_score = statistics.get("max_score", 0.0)
            sample_size = statistics.get("sample_size", 0)
            q25 = statistics.get("q25", 0.0)
            q75 = statistics.get("q75", 0.0)
            q90 = statistics.get("q90", 0.0)
        else:
            # Fallback for dataclass style
            trend_score = getattr(statistics, "trend_score", 0.0)
            median_score = getattr(statistics, "median_score", 0.0)
            std_score = getattr(statistics, "std_score", 0.0)
            min_score = getattr(statistics, "min_score", 0.0)
            max_score = getattr(statistics, "max_score", 0.0)
            sample_size = getattr(statistics, "sample_size", 0)
            q25 = getattr(statistics, "q25", 0.0)
            q75 = getattr(statistics, "q75", 0.0)
            q90 = getattr(statistics, "q90", 0.0)
        
        lines.append(f"  Trend Score:  {trend_score:.2f} / 100")
        
        # Determine regime based on trend score
        if trend_score >= 70:
            regime = "Strong Trend"
            detail = "High confidence trend environment"
        elif trend_score >= 55:
            regime = "Moderate Trend"
            detail = "Tendency toward directional movement"
        elif trend_score >= 40:
            regime = "Neutral"
            detail = "No strong directional bias"
        elif trend_score >= 25:
            regime = "Weak Trend"
            detail = "Slight tendency toward mean reversion"
        else:
            regime = "Range-Bound"
            detail = "Strong mean-reverting environment"
        
        lines.append(f"  Regime:       {regime}")
        lines.append(f"  Detail:       {detail}")
        lines.append(f"  Sample Size:  {sample_size:,}")

        lines.append("\n" + "-" * 45)
        lines.append("DISTRIBUTION STATISTICS")
        lines.append("-" * 45)
        lines.append(f"  Median:       {median_score:.2f}")
        lines.append(f"  Std Dev:      {std_score:.2f}")
        lines.append(f"  Min:          {min_score:.2f}")
        lines.append(f"  Max:          {max_score:.2f}")
        lines.append(f"  Q25:          {q25:.2f}")
        lines.append(f"  Q75:          {q75:.2f}")
        lines.append(f"  Q90:          {q90:.2f}")

        # Show weights if available
        weights = statistics.get("weights", {}) if isinstance(statistics, dict) else {}
        if weights:
            lines.append("\n" + "-" * 45)
            lines.append("COMPONENT WEIGHTS")
            lines.append("-" * 45)
            for component, weight in weights.items():
                lines.append(f"  {component}: {weight:.2f}")

        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)

        return "\n".join(lines)

    @staticmethod
    def save(report: str, filepath: str):
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
