def create_test_data(n=1000, seed=42):
    """Create synthetic test data with regimes."""
    np.random.seed(seed)
    
    # Generate price data with trends
    trend = np.cumsum(np.random.randn(n) * 0.5)
    close = 100 + trend
    
    high = close + np.abs(np.random.randn(n) * 1.0)
    low = close - np.abs(np.random.randn(n) * 1.0)
    volume = np.random.randn(n) * 100 + 1000
    
    # Create features with varying regime stability
    atr = np.random.randn(n) * 2 + 50
    rsi = np.random.randn(n) * 10 + 50
    trend_feature = np.random.randn(n) * 5 + 0
    
    # Target with some signal
    future_breakout = ((atr > 50) & (rsi > 50)).astype(int)
    
    df = pd.DataFrame({
        'open': 100 + np.random.randn(n) * 10,
        'high': high,
        'low': low,
        'close': close,
        'volume': volume,
        'ATR': atr,
        'RSI': rsi,
        'trend': trend_feature,
        'future_breakout': future_breakout
    })
    
    return df


def test_engine_trend_regime():
    """Test 1: Trend regime classification."""
    print("\n" + "=" * 60)
    print("TEST 1: Trend Regime Classification")
    print("=" * 60)
    
    df = create_test_data()
    engine = RegimeRobustnessEngine()
    
    trend = engine.compute_trend_regime(df)
    
    # Check all values are valid
    assert set(trend.unique()).issubset({'Bull', 'Bear', 'Sideways'})
    assert len(trend) == len(df)
    
    print(f"  Trend regimes: {trend.unique().tolist()}")
    print("  ✓ Trend regime test passed")
    return True
