import sys

import numpy as np
import pandas as pd

from atlas.modules.module104_conditional_expectancy.engine import ConditionalExpectancyEngine

from atlas.modules.module104_conditional_expectancy.validator import run_validation
from atlas.modules.module104_conditional_expectancy.module import ConditionalExpectancyResearch


def create_test_data(n=1000, seed=42):
    """
    Create synthetic test data similar to Module 103.
    
    Args:
        n: Number of samples
        seed: Random seed for reproducibility
        
    Returns:
        DataFrame with states and close prices
    """
    np.random.seed(seed)
    
    # Generate random states (0-15)
    states = np.random.randint(0, 16, n)
    
    # Generate close price with random walk
    close = 100 + np.cumsum(np.random.randn(n) * 0.5)
    
    df = pd.DataFrame({
        "state": states,
        "close": close
    })
    
    return df


def test_engine_future_returns():
    """Test 1: Future returns calculation."""
    print("\n" + "=" * 60)
    print("TEST 1: Future Returns Calculation")
    print("=" * 60)
    
    engine = ConditionalExpectancyEngine(future_horizon=5)
    
    # Create test close series
    close = pd.Series([100, 101, 102, 103, 104, 105, 106, 107, 108, 109, 110])
    
    future_returns = engine.compute_future_return(close)
    
    # Check shape
    assert len(future_returns) == len(close)
    
    # Check first value: (105/100 - 1) = 0.05
    assert np.isclose(future_returns[0], 0.05)
    
    # Check NaN at end (no future data)
    assert np.isnan(future_returns[-1])
    
    print(f"  Future returns shape: {future_returns.shape}")
    print(f"  First return: {future_returns[0]:.4f}")
    print("  ✓ Future returns test passed")
    return True


def test_engine_evaluate_state():
    """Test 2: State evaluation."""
    print("\n" + "=" * 60)
    print("TEST 2: State Evaluation")
    print("=" * 60)
    
    engine = ConditionalExpectancyEngine(future_horizon=5, min_samples=10)
    
    # Create test returns
    returns = np.array([0.01, 0.02, -0.01, 0.03, -0.02, 0.01, 0.02, -0.01, 0.03, -0.02])
    mask = np.array([True, True, True, True, True, True, True, True, True, True])
    
    result = engine.evaluate_state(returns, 5, mask)
    
    assert result is not None
    assert result["state"] == 5
    assert result["samples"] == 10
    assert result["win_rate"] == 0.6
    
    print(f"  State: {result['state']}")
    print(f"  Samples: {result['samples']}")
    print(f"  Win Rate: {result['win_rate']:.2f}")
    print("  ✓ State evaluation test passed")
    return True


def test_engine_compute():
    """Test 3: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Full Engine Computation")
    print("=" * 60)
    
    df = create_test_data(n=500)
    engine = ConditionalExpectancyEngine(future_horizon=5, min_samples=20)
    
    result = engine.compute(
        states=df["state"].values,
        close=df["close"]
    )
    
    # Check results
    assert result.total_states > 0, "Should have some states"
    assert result.sample_size > 0, "Should have samples"
    assert result.best_state >= 0, "Should have a best state"
    assert not result.expectancy_table.empty, "Expectancy table should not be empty"
    assert result.total_states <= 16, "Should not exceed 16 states"
    
    # Check table columns
    expected_cols = ["state", "samples", "mean", "median", "std", "min", "max", "win_rate"]
    for col in expected_cols:
        assert col in result.expectancy_table.columns, f"Missing column: {col}"
    
    print(f"  Total States: {result.total_states}")
    print(f"  Best State: {result.best_state}")
    print(f"  Sample Size: {result.sample_size}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 4: Statistics engine."""
    print("\n" + "=" * 60)
    print("TEST 4: Statistics Engine")
    print("=" * 60)
    
    df = create_test_data(n=500)
    engine = ConditionalExpectancyEngine(future_horizon=5, min_samples=20)
    stats_engine = ConditionalExpectancyStatisticsEngine()
    
    result = engine.compute(
        states=df["state"].values,
        close=df["close"]
    )
    
    stats = stats_engine.analyze(result)
    
    # Check statistics
    assert stats.total_states == result.total_states
    assert stats.best_state == result.best_state
    assert stats.sample_size == result.sample_size
    assert stats.positive_states + stats.negative_states == stats.total_states
    
    print(f"  Best State: {stats.best_state}")
    print(f"  Best Expectancy: {stats.best_expectancy:.2f}%")
    print(f"  Positive States: {stats.positive_states}")
    print(f"  Negative States: {stats.negative_states}")
    print("  ✓ Statistics test passed")
    return True


def test_validator():
    """Test 5: Validator."""
    print("\n" + "=" * 60)
    print("TEST 5: Validator")
    print("=" * 60)
    
    df = create_test_data(n=500)
    engine = ConditionalExpectancyEngine(future_horizon=5, min_samples=20)
    stats_engine = ConditionalExpectancyStatisticsEngine()
    
    result = engine.compute(
        states=df["state"].values,
        close=df["close"]
    )
    stats = stats_engine.analyze(result)
    
    passed = run_validation(result, stats)

    print("  ✓ Validator test passed")
    return True


def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)
    
    df = create_test_data(n=300)

def test_edge_cases():
    """Test 7: Edge cases."""
    print("\n" + "=" * 60)
    print("TEST 7: Edge Cases")
    print("=" * 60)
    
    engine = ConditionalExpectancyEngine(future_horizon=5, min_samples=100)
    
    # Test with insufficient samples
    df = create_test_data(n=50)
    result = engine.compute(
        states=df["state"].values,
        close=df["close"]
    )
    
    # May have fewer states due to min_samples threshold
    assert result.total_states >= 0
    assert result.sample_size >= 0
    
    # Test with all states
    df = create_test_data(n=10000)
    result = engine.compute(
        states=df["state"].values,
        close=df["close"]
    )
    
    # With enough samples, should see multiple states
    assert result.total_states > 0
    
    print(f"  Small dataset states: {result.total_states}")
    print("  ✓ Edge cases test passed")
    return True
