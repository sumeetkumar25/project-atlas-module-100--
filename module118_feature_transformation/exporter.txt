"""
Module 118 - Feature Transformation Exporter
"""

import pandas as pd
from pathlib import Path


class FeatureTransformationExporter:
    """Exporter for feature transformation results."""
    
    @staticmethod
    def export_transformed_dataframe(df: pd.DataFrame, filepath: str):
        """Export transformed dataframe to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_transformation_table(df: pd.DataFrame, filepath: str):
        """Export transformation table to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df.to_csv(path, index=False)
    
    @staticmethod
    def export_summary(summary: dict, filepath: str):
        """Export summary to CSV."""
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        df = pd.DataFrame([summary])
        df.to_csv(path, index=False)
