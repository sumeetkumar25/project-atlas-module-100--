def create_test_data(n=500, seed=42):
    """Create synthetic test data with various distributions."""
    np.random.seed(seed)
    
    # Normal distribution
    normal = np.random.randn(n) * 10 + 50
    
    # Skewed distribution
    skewed = np.random.exponential(2, n) * 10 + 10
    
    # Uniform bounded distribution
    bounded = np.random.uniform(0, 1, n)
    
    # Distribution with outliers
    outliers = np.random.randn(n) * 5 + 50
    outliers[:10] = 200  # Add outliers
    
    # Constant feature
    constant = np.ones(n) * 5
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'normal_feature': normal,
        'skewed_feature': skewed,
        'bounded_feature': bounded,
        'outlier_feature': outliers,
        'constant_feature': constant,
        'future_breakout': np.random.randint(0, 2, n)
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureTransformationEngine()
    
    features = engine.get_feature_columns(df)
    
    # Should detect these features
    assert 'normal_feature' in features
    assert 'skewed_feature' in features
    assert 'bounded_feature' in features
    assert 'outlier_feature' in features
    
    # Constant feature should be excluded (nunique == 1)
    assert 'constant_feature' not in features
    
    # OHLCV columns should be excluded
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    # Should only have 4 features (excludes constant_feature)
    assert len(features) == 4
    
    print(f"  Found {len(features)} features: {features}")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_analyze_distribution():

def test_engine_recommend_transform():
    """Test 3: Transformation recommendation."""
    print("\n" + "=" * 60)
    print("TEST 3: Transformation Recommendation")
    print("=" * 60)
    
    engine = FeatureTransformationEngine()
    
    # Normal distribution -> Z-Score
    normal_stats = {'skewness': 0.05, 'outlier_percentage': 1.0, 'n_unique': 100, 'min': -10, 'max': 10, 'std': 1}
    transform_name, strategy = engine.recommend_transform(normal_stats)
    assert transform_name == "Z-Score"
    
    # Highly skewed -> Log Transform
    skewed_stats = {'skewness': 2.0, 'outlier_percentage': 1.0, 'n_unique': 100, 'min': 0.1, 'max': 10, 'std': 1}
    transform_name, strategy = engine.recommend_transform(skewed_stats)
    assert transform_name == "Log Transform"
    
    # Many outliers -> Robust Scaler
    outlier_stats = {'skewness': 0.1, 'outlier_percentage': 10.0, 'n_unique': 100, 'min': -10, 'max': 10, 'std': 1}
    transform_name, strategy = engine.recommend_transform(outlier_stats)
    assert transform_name == "Robust Scaler"
    
    # Bounded -> Min-Max
    bounded_stats = {'skewness': 0.1, 'outlier_percentage': 1.0, 'n_unique': 100, 'min': 0, 'max': 1, 'std': 0.3}
    transform_name, strategy = engine.recommend_transform(bounded_stats)
    assert transform_name == "Min-Max Scaler"
    
    # Constant -> None
    constant_stats = {'skewness': 0.0, 'outlier_percentage': 0.0, 'n_unique': 1, 'min': 5, 'max': 5, 'std': 0}
    transform_name, strategy = engine.recommend_transform(constant_stats)
    assert transform_name == "None"
    
    print("  ✓ Transformation recommendation test passed")
    return True


def test_engine_zscore_transform():
    """Test 4: Z-score transformation."""
    print("\n" + "=" * 60)
    print("TEST 4: Z-Score Transformation")
    print("=" * 60)
    
    series = pd.Series(np.random.randn(100) * 10 + 50)
    strategy = ZScoreStrategy().fit(series)
    transformed = strategy.transform(series)
    
    assert abs(transformed.mean()) < 1e-10
    assert abs(transformed.std() - 1.0) < 0.1
    
    # Inverse transform
    inverse = strategy.inverse_transform(transformed)
    assert np.allclose(series.values, inverse.values, rtol=1e-10)
    
    print(f"  Mean: {transformed.mean():.6f}")
    print(f"  Std: {transformed.std():.6f}")
    print("  ✓ Z-score transform test passed")
    return True


def test_engine_robust_transform():
    """Test 5: Robust scaling."""
    print("\n" + "=" * 60)
    print("TEST 5: Robust Scaling")
    print("=" * 60)
    
    series = pd.Series(np.random.randn(100) * 10 + 50)
    series.iloc[0] = 200  # Add outlier
    strategy = RobustStrategy().fit(series)
    transformed = strategy.transform(series)
    
    assert abs(transformed.median()) < 1e-10
    assert abs(transformed.std() - 1.0) < 0.5
    
    # Inverse transform
    inverse = strategy.inverse_transform(transformed)
    assert np.allclose(series.values, inverse.values, rtol=1e-10)
    
    print("  ✓ Robust scaling test passed")
    return True


def test_engine_log_transform():
    """Test 6: Log transformation."""
    print("\n" + "=" * 60)
    print("TEST 6: Log Transformation")
    print("=" * 60)
    
    series = pd.Series(np.random.exponential(2, 100) * 10 + 10)
    strategy = LogStrategy().fit(series)
    transformed = strategy.transform(series)
    
    assert np.all(transformed > 0)
    
    # Inverse transform
    inverse = strategy.inverse_transform(transformed)
    assert np.allclose(series.values, inverse.values, rtol=1e-10)
    
    # Check skewness reduction
    before_skew = series.skew()
    after_skew = transformed.skew()
    assert after_skew < before_skew
    
    print(f"  Before skew: {before_skew:.4f}")
    print(f"  After skew: {after_skew:.4f}")
    print("  ✓ Log transform test passed")
    return True

    print("  ✓ Min-Max scaling test passed")

def test_transformation_reproducibility():
    """Test 13: Transformation reproducibility."""
    print("\n" + "=" * 60)

    print("=" * 60)
    
    df = create_test_data()
    selected_features = ['normal_feature', 'skewed_feature', 'bounded_feature', 'outlier_feature']
    
    # First run
    engine1 = FeatureTransformationEngine(selected_features=selected_features)
    result1 = engine1.compute(df)
    
    # Second run
    engine2 = FeatureTransformationEngine(selected_features=selected_features)
    result2 = engine2.compute(df)
    
    # Should produce identical results
    pd.testing.assert_frame_equal(
        result1.transformed_dataframe,
        result2.transformed_dataframe,
        check_dtype=False
    )
    pd.testing.assert_frame_equal(
        result1.transformation_table,
        result2.transformation_table,
        check_dtype=False
    )
    
    print("  ✓ Transformation reproducibility test passed")
    return True
