# atlas/modules/module118_feature_transformation/module.py
"""
Module 118
Feature Transformation

Converts KEEP features from Module 117 into standardized, model-ready variables.

Pipeline: load data → analyze distribution → recommend transforms →
          apply transforms → build transformed dataframe → statistics →
          validation → report → export
"""

from typing import Optional, Dict, Any, List
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureTransformationEngine
from .statistics import FeatureTransformationStatisticsEngine
from .validator import run_validation
from .report import FeatureTransformationReport
from .exporter import FeatureTransformationExporter


class FeatureTransformationResearch(BaseResearchModule):
    """Main research module for feature transformation."""
    
    MODULE_OUTPUT = "module118"
    METADATA = FeatureTransformationStatisticsEngine.METADATA
    
    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        selected_features: Optional[List[str]] = None,
        skew_threshold: float = 0.5,
        outlier_threshold: float = 1.5
    ):
        """
        Initialize the feature transformation module.
        
        Args:
            registry: Optional Atlas registry
            selected_features: List of features to transform (from Module 117 KEEP)
            skew_threshold: Threshold for recommending log transform
            outlier_threshold: IQR multiplier for outlier detection
        """
        super().__init__(
            module_name="Feature Transformation",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = FeatureTransformationEngine(
            selected_features=selected_features,
            skew_threshold=skew_threshold,
            outlier_threshold=outlier_threshold
        )
        self.stats_engine = FeatureTransformationStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.transformation_report = None
        self._output_dir = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_transformation", self._compute_transformation),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data from DataFrame."""
        return {"df": self.df}
    
    def _compute_transformation(self) -> Dict[str, Any]:
        """Step 1: Compute feature transformation."""
        self.logger.info("Step 1: Computing feature transformation")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            df=prepared["df"],
            selected_features=self.engine.selected_features
        )
        
        self.logger.metric(
            "Transformed Features",
            str(self.result_data.statistics["transformed_features"])
        )
        self.logger.metric(
            "Unchanged Features",
            str(self.result_data.statistics["unchanged_features"])
        )
        self.logger.metric(
            "Best Transform",
            self.result_data.statistics["best_transformation"]
        )
        self.logger.metric(
            "Avg Skew Before",
            f"{self.result_data.statistics['average_skew_before']:.4f}"
        )
        self.logger.metric(
            "Avg Skew After",
            f"{self.result_data.statistics['average_skew_after']:.4f}"
        )
        self.logger.success("Feature transformation computed")
        
        # Generate report
        self.transformation_report = FeatureTransformationReport.generate_transformation_table(
            self.result_data.transformation_table
        )
        
        return {
            "transformed_features": self.result_data.statistics["transformed_features"],
            "unchanged_features": self.result_data.statistics["unchanged_features"],
            "best_transform": self.result_data.statistics["best_transformation"],
            "avg_skew_before": self.result_data.statistics["average_skew_before"],
            "avg_skew_after": self.result_data.statistics["average_skew_after"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Total Features",
            str(self.statistics.total_features)
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["transformed_dataframe"] = self.result_data.transformed_dataframe
        self.result.artifacts["transformation_table"] = self.result_data.transformation_table
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary])
        self.result.artifacts["transformation_summary"] = summary_df
        
        # Generate and register reports
        main_report = FeatureTransformationReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_transformation_report"] = main_report
        self.result.artifacts["transformation_report"] = self.transformation_report
        
        # Export to files
        if self._output_dir is not None:
            export_dir = self._output_dir / "exports"
            export_dir.mkdir(parents=True, exist_ok=True)
            
            FeatureTransformationExporter.export_transformed_dataframe(
                self.result_data.transformed_dataframe,
                export_dir / "transformed_dataframe.csv"
            )
            
            FeatureTransformationExporter.export_transformation_table(
                self.result_data.transformation_table,
                export_dir / "transformation_table.csv"
            )
            
            FeatureTransformationExporter.export_summary(
                self.result_data.summary,
                export_dir / "transformation_summary.csv"
            )
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "transformed_features": self.result_data.statistics["transformed_features"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Transformation Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 118 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
