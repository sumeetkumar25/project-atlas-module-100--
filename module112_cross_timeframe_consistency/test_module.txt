def create_test_datasets():
    """Create synthetic test datasets for multiple timeframes."""
    np.random.seed(42)
    
    datasets = {}
    timeframes = ["15m", "1h", "4h"]
    n_samples = {"15m": 500, "1h": 300, "4h": 150}
    
    for tf in timeframes:
        n = n_samples[tf]
        
        # Create features
        atr = np.random.randn(n) * 2 + 50
        rsi = np.random.randn(n) * 10 + 50
        volume = np.random.randn(n) * 100 + 1000
        trend = np.random.randn(n) * 5 + 0
        
        # Create target
        future_breakout = ((atr > 50) & (rsi > 50)).astype(int)
        
        df = pd.DataFrame({
            "timestamp": pd.date_range("2020-01-01", periods=n, freq="1min"),
            "open": 100 + np.random.randn(n) * 10,
            "high": 105 + np.random.randn(n) * 10,
            "low": 95 + np.random.randn(n) * 10,
            "close": 100 + np.random.randn(n) * 10,
            "volume": volume,
            "ATR": atr,
            "RSI": rsi,
            "trend": trend,
            "future_breakout": future_breakout
        })
        
        datasets[tf] = df
    
    return datasets

def test_engine_stability_scores():
    """Test 6: Stability scores computation."""
    print("\n" + "=" * 60)
    print("TEST 6: Stability Scores")
    print("=" * 60)

    engine = CrossTimeframeConsistencyEngine()
    datasets = create_test_datasets()
    features = engine.get_common_features(datasets)

    feature_stats = engine.compute_feature_statistics(
        datasets,
        features
    )

    distribution_consistency = (
        engine.compute_distribution_consistency(
            datasets,
            features
        )
    )

    correlations = engine.compute_correlation_consistency(
        datasets,
        features
    )

    (
        rankings,
        spearman,
        kendall,
        feature_rank_stability
    ) = engine.compute_ranking_consistency(
        datasets,
        correlations,
        ["15m", "1h", "4h"]
    )

    stability = engine.compute_stability_scores(
        feature_stats,
        correlations,
        distribution_consistency,
        feature_rank_stability
    )

    assert not stability.empty
    assert "stability_score" in stability.columns

    assert all(
        0 <= score <= 100
        for score in stability["stability_score"]
    )

    print(f"  Stability scores shape: {stability.shape}")
    print(
        f"  Mean stability: "
        f"{stability['stability_score'].mean():.2f}"
    )
    print("  ✓ Stability scores test passed")

    return True
