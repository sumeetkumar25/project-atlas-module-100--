# atlas/modules/module106_state_transition/module.py
"""
Module 106
State Transition Matrix

Analyzes how market states transition from one state to another.

Pipeline: load states → build transition matrix → calculate probabilities → statistics → validation → report → export
"""

from typing import Optional, Dict, Any
import pandas as pd
import numpy as np

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import StateTransitionEngine
from .statistics import StateTransitionStatisticsEngine
from .validator import run_validation
from .report import StateTransitionReport


class StateTransitionResearch(BaseResearchModule):
    """Main research module for state transition analysis."""
    
    MODULE_OUTPUT = "module106"
    METADATA = StateTransitionStatisticsEngine.METADATA
    
    def __init__(self, registry: Optional[AtlasRegistry] = None):
        """
        Initialize the state transition module.
        
        Args:
            registry: Optional Atlas registry
        """
        super().__init__(
            module_name="State Transition Matrix",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = StateTransitionEngine()
        self.stats_engine = StateTransitionStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.transition_report = None
    
    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("state_transition", self._compute_transition),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data from DataFrame."""
        # Check if we have states column (from Module 103)
        if "state" not in self.df.columns:
            raise ValueError("DataFrame must contain 'state' column (from Module 103)")
        
        return {
            "states": self.df["state"].values
        }
    
    def _compute_transition(self) -> Dict[str, Any]:
        """Step 1: Compute state transitions."""
        self.logger.info("Step 1: Computing state transitions")
        
        prepared = self._prepare_data()
        
        self.result_data = self.engine.compute(
            states=prepared["states"]
        )
        
        self.logger.metric(
            "Self-Transition Rate",
            f"{self.result_data.self_transition_rate:.3f}"
        )
        self.logger.metric(
            "Entropy",
            f"{self.result_data.entropy:.3f} bits"
        )
        self.logger.metric(
            "Sample Size",
            f"{self.result_data.sample_size:,}"
        )
        self.logger.success("State transitions computed")
        
        # Generate transition report
        self.transition_report = StateTransitionReport.generate_transition_table(
            self.result_data.transition_probabilities
        )
        
        return {
            "self_transition_rate": self.result_data.self_transition_rate,
            "entropy": self.result_data.entropy,
            "sample_size": self.result_data.sample_size,
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Compute statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Deterministic States",
            f"{self.statistics.deterministic_states}/16"
        )
        self.logger.metric(
            "Random States",
            f"{self.statistics.random_states}/16"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Store transition matrices
        self.result.artifacts["transition_counts"] = self.result_data.transition_counts
        self.result.artifacts["transition_probabilities"] = self.result_data.transition_probabilities
        
        # Store main report
        main_report = StateTransitionReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["state_transition_report"] = main_report
        
        # Store transition report
        self.result.artifacts["transition_report"] = self.transition_report
        
        # Store detailed results as dict
        self.result.artifacts["detailed_results"] = {
            "transition_counts": self.result_data.transition_counts.to_dict(),
            "transition_probabilities": self.result_data.transition_probabilities.to_dict(),
            "self_transition_rate": self.result_data.self_transition_rate,
            "entropy": self.result_data.entropy,
            "sample_size": self.result_data.sample_size
        }
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("State Transition Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 106 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on DataFrame."""
        return super().run_dataframe(df, dataset_id, output_dir)
