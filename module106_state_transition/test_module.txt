def create_test_data(n=1000, seed=42):
    """
    Create synthetic test data with state transitions.
    
    Args:
        n: Number of samples
        seed: Random seed for reproducibility
        
    Returns:
        DataFrame with states
    """
    np.random.seed(seed)
    
    # Create states with some persistence
    states = []
    current_state = np.random.randint(0, 16)
    
    for _ in range(n):
        # 70% chance to stay in same state, 30% to change
        if np.random.random() < 0.7:
            states.append(current_state)
        else:
            current_state = np.random.randint(0, 16)
            states.append(current_state)
    
    df = pd.DataFrame({
        "state": states
    })
    
    return df


def test_engine_count_matrix():
    """Test 1: Count matrix building."""
    print("\n" + "=" * 60)
    print("TEST 1: Count Matrix Building")
    print("=" * 60)
    
    engine = StateTransitionEngine()
    
    # Test with known sequence
    states = np.array([4, 4, 4, 7, 7, 2, 2, 13, 13, 13, 5])
    
    counts = engine.build_count_matrix(states)
    
    # Check shape
    assert counts.shape == (16, 16)
    
    # Check specific transitions
    assert counts.iloc[4, 4] == 2  # 4->4
    assert counts.iloc[4, 7] == 1  # 4->7
    assert counts.iloc[7, 7] == 1  # 7->7
    assert counts.iloc[7, 2] == 1  # 7->2
    assert counts.iloc[2, 2] == 1  # 2->2
    assert counts.iloc[2, 13] == 1  # 2->13
    assert counts.iloc[13, 13] == 2  # 13->13
    assert counts.iloc[13, 5] == 1  # 13->5
    
    print("  Count matrix built successfully")
    print(f"  Total transitions: {counts.sum().sum()}")
    print("  ✓ Count matrix test passed")
    return True


def test_engine_probability_matrix():
    """Test 2: Probability matrix building."""
    print("\n" + "=" * 60)
    print("TEST 2: Probability Matrix Building")
    print("=" * 60)
    
    engine = StateTransitionEngine()
    
    states = np.array([4, 4, 4, 7, 7, 2, 2, 13, 13, 13, 5])
    counts = engine.build_count_matrix(states)
    prob = engine.build_probability_matrix(counts)
    
    # Check shape
    assert prob.shape == (16, 16)
    
    # Check rows sum to 1 for states with transitions
    row_sums = prob.sum(axis=1)
    has_transitions = counts.sum(axis=1) > 0
    assert np.allclose(row_sums[has_transitions], 1)
    
    # Check specific probabilities
    # State 4: 2 transitions to 4, 1 to 7 → p(4→4)=2/3, p(4→7)=1/3
    assert np.isclose(prob.iloc[4, 4], 2/3)
    assert np.isclose(prob.iloc[4, 7], 1/3)
    
    # State 7: 1 to 7, 1 to 2 → p(7→7)=1/2, p(7→2)=1/2
    assert np.isclose(prob.iloc[7, 7], 0.5)
    assert np.isclose(prob.iloc[7, 2], 0.5)
    
    print("  Probability matrix built successfully")
    print(f"  Row sums for state 4: {prob.iloc[4].sum():.3f}")
    print(f"  Row sums for state 7: {prob.iloc[7].sum():.3f}")
    print("  ✓ Probability matrix test passed")
    return True


def test_engine_self_transition():
    """Test 3: Self-transition rate calculation."""
    print("\n" + "=" * 60)
    print("TEST 3: Self-Transition Rate")
    print("=" * 60)
    
    engine = StateTransitionEngine()
    
    states = np.array([4, 4, 4, 7, 7, 2, 2, 13, 13, 13, 5])
    counts = engine.build_count_matrix(states)
    self_rate = engine.calculate_self_transition_rate(counts)
    
    # Total transitions = 10 (11 states → 10 transitions)
    # Self-transitions: 4→4 (2), 7→7 (1), 2→2 (1), 13→13 (2) = 6
    # Rate = 6/10 = 0.6
    assert np.isclose(self_rate, 0.6)
    
    print(f"  Self-transition rate: {self_rate:.3f}")
    print("  ✓ Self-transition test passed")
    return True


def test_engine_entropy():
    """Test 4: Entropy calculation."""
    print("\n" + "=" * 60)
    print("TEST 4: Transition Entropy")
    print("=" * 60)
    
    engine = StateTransitionEngine()
    
    states = np.array([4, 4, 4, 7, 7, 2, 2, 13, 13, 13, 5])
    counts = engine.build_count_matrix(states)
