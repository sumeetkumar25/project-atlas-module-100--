def create_test_data(n=1000, seed=42):
    """Create synthetic test data with degrading feature."""
    np.random.seed(seed)
    
    # Split into 5 windows
    window_size = n // 5
    
    features_data = []
    target_data = []
    
    # Feature that degrades over time
    auc_values = [0.82, 0.80, 0.75, 0.69, 0.61]
    
    for window_idx, auc in enumerate(auc_values):
        start = window_idx * window_size
        end = start + window_size if window_idx < 4 else n
        
        # Generate feature with degrading AUC
        n_samples = end - start
        
        # Generate feature with signal based on AUC
        # AUC ~0.8 means strong signal, AUC ~0.6 means weaker signal
        signal_strength = (auc - 0.5) * 2  # 0.5->0, 0.8->0.6, 0.6->0.2
        feature = np.random.randn(n_samples)
        target_prob = 1 / (1 + np.exp(-(signal_strength * feature)))
        target = (np.random.random(n_samples) < target_prob).astype(int)
        
        features_data.extend(feature.tolist())
        target_data.extend(target.tolist())
    
    # Create another feature that is stable
    stable_feature = np.random.randn(n)
    stable_target_prob = 1 / (1 + np.exp(-(0.3 * stable_feature)))
    stable_target = (np.random.random(n) < stable_target_prob).astype(int)
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.cumsum(np.random.randn(n) * 0.5),
        'high': 100 + np.cumsum(np.random.randn(n) * 0.5) + np.abs(np.random.randn(n) * 0.5),
        'low': 100 + np.cumsum(np.random.randn(n) * 0.5) - np.abs(np.random.randn(n) * 0.5),
        'close': 100 + np.cumsum(np.random.randn(n) * 0.5),
        'volume': np.random.randn(n) * 100 + 1000,
        'degrading_feature': features_data,
        'stable_feature': stable_feature,
        'future_breakout': target_data
    })
    
    return df


def test_engine_windows():
    """Test 1: Window creation."""
    print("\n" + "=" * 60)
    print("TEST 1: Window Creation")
    print("=" * 60)
    
    df = create_test_data()
    engine = TemporalStabilityEngine(n_windows=5)
    
    windows = engine.create_windows(df)
    
    assert len(windows) == 5
    assert sum(len(w) for w in windows) == len(df)
    
    print(f"  Created {len(windows)} windows")
    print("  ✓ Window creation test passed")
    return True


def test_engine_features():
    """Test 2: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 2: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = TemporalStabilityEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'degrading_feature' in features
    assert 'stable_feature' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Features: {features}")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_window_scores():
    """Test 3: Window scores."""
    print("\n" + "=" * 60)
    print("TEST 3: Window Scores")
    print("=" * 60)
    
    df = create_test_data()
    engine = TemporalStabilityEngine(n_windows=5, min_samples=10)
    
    windows = engine.create_windows(df)
    features = engine.get_feature_columns(df)
    
    window_scores = engine.compute_window_scores(windows, features)
    
    assert not window_scores.empty
    assert 'window' in window_scores.columns
    assert 'feature' in window_scores.columns
    assert 'score' in window_scores.columns
    
    print(f"  Window scores: {len(window_scores)} rows")
    print("  ✓ Window scores test passed")
    return True


def test_engine_temporal_statistics():
    """Test 4: Temporal statistics."""
    print("\n" + "=" * 60)
    print("TEST 4: Temporal Statistics")
    print("=" * 60)
    
    df = create_test_data()
    engine = TemporalStabilityEngine(n_windows=5, min_samples=10)
    
    windows = engine.create_windows(df)
    features = engine.get_feature_columns(df)
    window_scores = engine.compute_window_scores(windows, features)
    stats = engine.compute_temporal_statistics(window_scores)
    
    assert not stats.empty
    assert 'feature' in stats.columns
    assert 'temporal_score' in stats.columns
    assert 'slope' in stats.columns
    assert 'decay' in stats.columns
    
    # Stable feature should have higher temporal score
    stable_score = stats[stats['feature'] == 'stable_feature']['temporal_score'].iloc[0] if not stats[stats['feature'] == 'stable_feature'].empty else 0
    degrading_score = stats[stats['feature'] == 'degrading_feature']['temporal_score'].iloc[0] if not stats[stats['feature'] == 'degrading_feature'].empty else 0
    
    print(f"  Stable feature score: {stable_score:.2f}")
    print(f"  Degrading feature score: {degrading_score:.2f}")
    print("  ✓ Temporal statistics test passed")
    return True


def test_engine_slope_detection():
    """Test 5: Trend drift detection."""
    print("\n" + "=" * 60)
    print("TEST 5: Trend Drift Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = TemporalStabilityEngine(n_windows=5, min_samples=10)
    
    windows = engine.create_windows(df)
    features = engine.get_feature_columns(df)
    window_scores = engine.compute_window_scores(windows, features)
    stats = engine.compute_temporal_statistics(window_scores)
    
    # Degrading feature should have negative slope
    degrading_slope = stats[stats['feature'] == 'degrading_feature']['slope'].iloc[0] if not stats[stats['feature'] == 'degrading_feature'].empty else 0
    stable_slope = stats[stats['feature'] == 'stable_feature']['slope'].iloc[0] if not stats[stats['feature'] == 'stable_feature'].empty else 0
    
    assert degrading_slope < 0
    assert abs(degrading_slope) > abs(stable_slope)
    
    print(f"  Degrading slope: {degrading_slope:.4f}")
    print(f"  Stable slope: {stable_slope:.4f}")
    print("  ✓ Trend drift test passed")
    return True

    """Test 6: Full engine computation."""
