def create_test_data(n=500, seed=42):
    """Create synthetic test data with redundant features."""
    np.random.seed(seed)
    
    # Base features
    x1 = np.random.randn(n)
    x2 = np.random.randn(n) * 0.5 + x1 * 0.5  # Correlated with x1
    x3 = np.random.randn(n) * 0.2 + x1 * 0.8  # Highly correlated with x1
    x4 = np.random.randn(n)  # Independent
    x5 = np.random.randn(n) * 0.5 + x4 * 0.5  # Correlated with x4
    x6 = np.random.randn(n)  # Independent
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'feature5': x5,
        'feature6': x6,
        'future_breakout': np.random.randint(0, 2, n)
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'feature1' in features
    assert 'feature2' in features
    assert 'feature3' in features
    assert 'feature4' in features
    assert 'feature5' in features
    assert 'feature6' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True

def test_engine_clustering():
    """Test 8: Feature clustering."""
    print("\n" + "=" * 60)
    print("TEST 8: Feature Clustering")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureRedundancyEngine(cluster_threshold=0.40)
    
    features = engine.get_feature_columns(df)
    n = len(features)
    redundancy_matrix = np.eye(n)
    
    # Create synthetic redundancy matrix
    for i in range(n):
        for j in range(i+1, n):
            if i < 3 and j < 3:  # First three features are redundant
                redundancy_matrix[i, j] = 0.8
                redundancy_matrix[j, i] = 0.8
            else:
                redundancy_matrix[i, j] = 0.1
                redundancy_matrix[j, i] = 0.1
    
    redundancy_matrix = pd.DataFrame(redundancy_matrix, index=features, columns=features)
    clusters = engine.cluster_features(df, features, redundancy_matrix)
    
    assert not clusters.empty
    assert 'feature' in clusters.columns
    assert 'cluster' in clusters.columns
    
    # Feature1, feature2, feature3 should be in same cluster
    cluster1 = clusters[clusters['feature'] == 'feature1']['cluster'].iloc[0]
    cluster2 = clusters[clusters['feature'] == 'feature2']['cluster'].iloc[0]
    cluster3 = clusters[clusters['feature'] == 'feature3']['cluster'].iloc[0]
    assert cluster1 == cluster2 == cluster3
    
    print(f"  Found {clusters['cluster'].nunique()} clusters")
    print("  ✓ Clustering test passed")
    return True
