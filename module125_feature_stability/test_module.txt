def create_test_data(n=1000, seed=42):
    """Create synthetic test data with known stability behavior."""
    np.random.seed(seed)
    
    # Feature 1: Stable (normal distribution)
    stable_feature = np.random.normal(0, 1, n)
    
    # Feature 2: Slow drift (linear trend + noise)
    drift = np.linspace(0, 2, n)
    slow_drift_feature = drift + np.random.normal(0, 0.5, n)
    
    # Feature 3: Strong drift (shift in distribution)
    strong_drift_feature = np.zeros(n)
    half = n // 2
    strong_drift_feature[:half] = np.random.normal(0, 1, half)
    strong_drift_feature[half:] = np.random.normal(5, 1, n - half)
    
    # Feature 4: Medium drift (gradual shift)
    medium_drift_feature = np.zeros(n)
    for i in range(n):
        medium_drift_feature[i] = np.random.normal(i / n * 3, 0.8)
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'stable_feature': stable_feature,
        'slow_drift_feature': slow_drift_feature,
        'medium_drift_feature': medium_drift_feature,
        'strong_drift_feature': strong_drift_feature,
        'future_breakout': np.random.randint(0, 2, n)
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityAnalysisEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'stable_feature' in features
    assert 'slow_drift_feature' in features
    assert 'medium_drift_feature' in features
    assert 'strong_drift_feature' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_windows():
    """Test 2: Window creation."""
    print("\n" + "=" * 60)
    print("TEST 2: Window Creation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityAnalysisEngine(n_windows=5)
    
    windows = engine.create_windows(df)
    
    assert len(windows) == 5
    assert sum(len(w) for w in windows) == len(df)
    
    print(f"  Created {len(windows)} windows")
    print("  ✓ Window creation test passed")
    return True


def test_engine_mean_stability():
    """Test 3: Mean stability computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Mean Stability")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityAnalysisEngine(n_windows=5, min_samples_per_window=10)
    
    # Stable feature should have high mean stability
    stable = df['stable_feature']
    stable_score = engine.compute_mean_stability(stable)
    
    # Unstable feature should have low mean stability
    strong = df['strong_drift_feature']
    strong_score = engine.compute_mean_stability(strong)
    
    assert stable_score > strong_score
    assert 0 <= stable_score <= 1
    assert 0 <= strong_score <= 1
    
    print(f"  Stable mean stability: {stable_score:.3f}")
    print(f"  Unstable mean stability: {strong_score:.3f}")
    print("  ✓ Mean stability test passed")
    return True

def test_engine_decisions():
    """Test 7: Decision making."""
    print("\n" + "=" * 60)
    print("TEST 7: Decision Making")
    print("=" * 60)
    
    engine = FeatureStabilityAnalysisEngine(
        pass_threshold=0.70,
        warning_threshold=0.50
    )
    
    assert engine.make_decision(0.85) == "PASS"
    assert engine.make_decision(0.70) == "PASS"
    assert engine.make_decision(0.60) == "WARNING"
    assert engine.make_decision(0.50) == "WARNING"
    assert engine.make_decision(0.45) == "FAIL"
    
    print("  ✓ Decision making test passed")
    return True


def test_engine_compute():
    """Test 8: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 8: Full Engine Computation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityAnalysisEngine(n_windows=5, min_samples_per_window=10)
    
    result = engine.compute(df)
    
    assert not result.stability_table.empty
    assert len(result.stable_features) > 0
    assert len(result.unstable_features) > 0
    assert (
        len(result.stable_features)
        + len(result.unstable_features)
        == result.summary_statistics["total_features"]
    )
    assert result.summary_statistics["average_stability"] > 0
    
    # Stable feature should be PASS
    stable_row = result.stability_table[result.stability_table['feature'] == 'stable_feature']
    if not stable_row.empty:
        assert stable_row['decision'].iloc[0] == "PASS"
    
    # Strong drift feature should be FAIL
    strong_row = result.stability_table[result.stability_table['feature'] == 'strong_drift_feature']
    if not strong_row.empty:
        assert strong_row['decision'].iloc[0] == "FAIL"
    
    print(f"  STABLE: {len(result.stable_features)}")
    print(f"  UNSTABLE: {len(result.unstable_features)}")
    print(f"  Avg Stability: {result.summary_statistics['average_stability']:.3f}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 9: Statistics engine."""

    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityAnalysisEngine(n_windows=5, min_samples_per_window=10)

    print(f"  Total: {stats.total_features}")
    print(f"  STABLE: {stats.stable_count}")
    print(f"  UNSTABLE: {stats.unstable_count}")
    print("  ✓ Statistics test passed")
    return True
