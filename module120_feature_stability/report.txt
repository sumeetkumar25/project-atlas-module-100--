"""
Module 120 - Feature Stability Report
"""

from datetime import datetime
import pandas as pd


class FeatureStabilityReport:
    """Report generator for feature stability results."""
    
    @staticmethod
    def generate(statistics, metadata=None, dataset_name="Unknown") -> str:
        """Generate a formatted report."""
        lines = []
        
        lines.append("=" * 55)
        lines.append("MODULE 120")
        lines.append("FEATURE STABILITY")
        lines.append("=" * 55)
        lines.append(f"\nDataset: {dataset_name}")
        
        # Overview
        lines.append("\n" + "-" * 45)
        lines.append("OVERVIEW")
        lines.append("-" * 45)
        lines.append(f"  Total Features: {statistics.total_features}")
        lines.append(f"  PASS: {statistics.passed_count}")
        lines.append(f"  WARNING: {statistics.warning_count}")
        lines.append(f"  FAIL: {statistics.failed_count}")
        
        # Percentages
        lines.append("\n" + "-" * 45)
        lines.append("PERCENTAGES")
        lines.append("-" * 45)
        lines.append(f"  PASS: {statistics.passed_percentage:.1f}%")
        lines.append(f"  WARNING: {statistics.warning_percentage:.1f}%")
        lines.append(f"  FAIL: {statistics.failed_percentage:.1f}%")
        
        # Stability Scores
        lines.append("\n" + "-" * 45)
        lines.append("STABILITY SCORES")
        lines.append("-" * 45)
        lines.append(f"  Average: {statistics.average_stability:.3f}")
        lines.append(f"  Median: {statistics.median_stability:.3f}")
        lines.append(f"  Best: {statistics.best_score:.3f}")
        lines.append(f"  Worst: {statistics.worst_score:.3f}")
        
        # Best and Worst Features
        if statistics.best_feature:
            lines.append("\n" + "-" * 45)
            lines.append("BEST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.best_feature}: {statistics.best_score:.3f}")
        
        if statistics.worst_feature:
            lines.append("\n" + "-" * 45)
            lines.append("WORST FEATURE")
            lines.append("-" * 45)
            lines.append(f"  {statistics.worst_feature}: {statistics.worst_score:.3f}")
        
        # Drift Metrics
        lines.append("\n" + "-" * 45)
        lines.append("AVERAGE DRIFT")
        lines.append("-" * 45)
        lines.append(f"  Mean Drift: {statistics.average_drift:.4f}")
        lines.append(f"  KS Drift: {statistics.average_ks:.4f}")
        lines.append(f"  Entropy Drift: {statistics.average_entropy_drift:.4f}")
        
        # Interpretation
        lines.append("\n" + "-" * 45)
        lines.append("INTERPRETATION")
        lines.append("-" * 45)
        
        if statistics.average_stability > 0.8:
            lines.append("  ✅ Features are highly stable over time")
        elif statistics.average_stability > 0.6:
            lines.append("  ✓ Features show moderate stability")
        else:
            lines.append("  ⚠ Features show low temporal stability")
        
        if statistics.average_drift < 0.1:
            lines.append("  ✅ Low average drift")
        elif statistics.average_drift < 0.3:
            lines.append("  ✓ Moderate average drift")
        else:
            lines.append("  ⚠ High average drift")
        
        # Validation
        lines.append("\n" + "-" * 45)
        lines.append("VALIDATION")
        lines.append("-" * 45)
        lines.append("  Status: PASSED")
        
        # Footer
        lines.append(f"\n  Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
        lines.append("\n" + "=" * 55)
        lines.append("END OF REPORT")
        lines.append("=" * 55)
        
        return "\n".join(lines)
    
    @staticmethod
    def generate_stability_table(stability_table: pd.DataFrame, top_n: int = 10) -> str:
        """Generate a stability table report."""
        lines = []
        lines.append("\n" + "=" * 80)
        lines.append(f"TOP {top_n} FEATURES BY STABILITY")
        lines.append("=" * 80)
        
        top_features = stability_table.head(top_n)
        
        lines.append(
            f"\n{'Feature':<20} {'Score':<10} {'Decision':<12} {'Mean Drift':<12} {'KS Drift':<12} {'Entropy':<12}"
        )
        lines.append("-" * 80)
        
        for _, row in top_features.iterrows():
            lines.append(
                f"{row['feature'][:19]:<20} "
                f"{row['stability_score']:<10.3f} "
                f"{row['decision']:<12} "
                f"{row['mean_drift']:<12.4f} "
                f"{row['ks_drift']:<12.4f} "
                f"{row['entropy_drift']:<12.4f}"
            )
        
        lines.append("\n" + "=" * 80)
        return "\n".join(lines)
    
    @staticmethod
    def save(report: str, filepath: str):
        """Save report to file."""
        from pathlib import Path
        path = Path(filepath)
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            f.write(report)
