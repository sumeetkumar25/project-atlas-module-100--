def create_test_data(n=1000, seed=42):
    """Create synthetic test data with various stability levels."""
    np.random.seed(seed)
    
    # Stable feature (constant mean)
    stable = np.random.randn(n) * 10 + 50
    
    # Drifting feature (mean changes over time)
    drift = np.random.randn(n) * 10 + 50 + np.linspace(0, 20, n)
    
    # Feature with increasing variance
    variance_drift = np.random.randn(n) * (10 + np.linspace(0, 5, n)) + 50
    
    # Feature with outliers
    outliers = np.random.randn(n) * 10 + 50
    outliers[:20] = 500
    
    # Feature with missing values
    missing_feature = np.random.randn(n) * 10 + 50
    missing_feature[:100] = np.nan
    
    df = pd.DataFrame({
        'timestamp': pd.date_range('2020-01-01', periods=n, freq='1h'),
        'open': 100 + np.random.randn(n) * 10,
        'high': 100 + np.random.randn(n) * 10 + np.abs(np.random.randn(n) * 2),
        'low': 100 + np.random.randn(n) * 10 - np.abs(np.random.randn(n) * 2),
        'close': 100 + np.random.randn(n) * 10,
        'volume': np.random.randn(n) * 100 + 1000,
        'stable_feature': stable,
        'drifting_feature': drift,
        'variance_drift_feature': variance_drift,
        'outlier_feature': outliers,
        'missing_feature': missing_feature,
        'future_breakout': np.random.randint(0, 2, n)
    })
    
    return df


def test_engine_feature_detection():
    """Test 1: Feature detection."""
    print("\n" + "=" * 60)
    print("TEST 1: Feature Detection")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityEngine()
    
    features = engine.get_feature_columns(df)
    
    assert 'stable_feature' in features
    assert 'drifting_feature' in features
    assert 'variance_drift_feature' in features
    assert 'outlier_feature' in features
    assert 'missing_feature' in features
    assert 'close' not in features
    assert 'future_breakout' not in features
    
    print(f"  Found {len(features)} features")
    print("  ✓ Feature detection test passed")
    return True


def test_engine_mean_stability():
    """Test 2: Mean stability computation."""
    print("\n" + "=" * 60)
    print("TEST 2: Mean Stability")
    print("=" * 60)
    
    engine = FeatureStabilityEngine(window=50)
    
    # Stable series
    stable = pd.Series(np.random.randn(1000) * 10 + 50)
    stable_mean = engine.compute_mean_stability(stable)
    
    # Drifting series
    drift = pd.Series(np.random.randn(1000) * 10 + 50 + np.linspace(0, 10, 1000))
    drift_mean = engine.compute_mean_stability(drift)
    
    # Drifting series should have higher mean drift
    assert drift_mean["mean_drift"] > stable_mean["mean_drift"]
    
    print(f"  Stable mean drift: {stable_mean['mean_drift']:.4f}")
    print(f"  Drifting mean drift: {drift_mean['mean_drift']:.4f}")
    print("  ✓ Mean stability test passed")
    return True


def test_engine_variance_stability():
    """Test 3: Variance stability computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Variance Stability")
    print("=" * 60)
    
    engine = FeatureStabilityEngine(window=50)
    
    # Stable variance (constant variance)
    stable = pd.Series(np.random.randn(1000) * 10 + 50)
    stable_var = engine.compute_variance_stability(stable)
    
    # Unstable variance (increasing variance over time)
    unstable = pd.Series(np.random.randn(1000) * (10 + np.linspace(0, 10, 1000)) + 50)
    unstable_var = engine.compute_variance_stability(unstable)
    
    # Unstable variance should have higher drift than stable variance
    assert unstable_var["mean_drift"] > stable_var["mean_drift"]
    
    print(f"  Stable variance drift: {stable_var['mean_drift']:.4f}")
    print(f"  Unstable variance drift: {unstable_var['mean_drift']:.4f}")
    print("  ✓ Variance stability test passed")
    return True


def test_engine_ks_drift():
    """Test 4: KS drift computation."""
    print("\n" + "=" * 60)
    print("TEST 4: KS Drift")
    print("=" * 60)
    
    engine = FeatureStabilityEngine(window=50)
    
    # Same distribution
    series1 = pd.Series(np.random.randn(1000) * 10 + 50)
    ks1 = engine.compute_ks_drift(series1)
    
    # Different distributions (first half vs second half)
    first_half = np.random.randn(500) * 10 + 50
    second_half = np.random.randn(500) * 20 + 50
    series2 = pd.Series(np.concatenate([first_half, second_half]))
    ks2 = engine.compute_ks_drift(series2)
    
    # Different distributions should have higher KS drift
    assert ks2['ks_drift'] > ks1['ks_drift']
    
    print(f"  Same distribution KS drift: {ks1['ks_drift']:.4f}")
    print(f"  Different distribution KS drift: {ks2['ks_drift']:.4f}")
    print("  ✓ KS drift test passed")
    return True

def test_engine_decisions():
    """Test 7: Decision making."""
    print("\n" + "=" * 60)
    print("TEST 7: Decision Making")
    print("=" * 60)
    
    engine = FeatureStabilityEngine(
        pass_threshold=0.80,
        warning_threshold=0.60
    )
    
    assert engine.make_decision(0.90) == "PASS"
    assert engine.make_decision(0.70) == "WARNING"
    assert engine.make_decision(0.50) == "FAIL"
    assert engine.make_decision(0.80) == "PASS"

    assert engine.make_decision(0.59) == "FAIL"
    
    print("  ✓ Decision making test passed")

def test_engine_compute():
    """Test 8: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 8: Full Engine Computation")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityEngine(window=50)
    
    result = engine.compute(df)
    
    # Verify stability table exists
    assert not result.stability_table.empty
    
    # Verify every feature received exactly one decision
    assert len(result.stability_table) == 5  # 5 features in synthetic dataset
    
    # Verify all decisions are valid
    decisions = set(result.stability_table["decision"])
    assert decisions.issubset({"PASS", "WARNING", "FAIL"})
    
    # Verify the sum of decisions equals total features
    assert (
        len(result.passed_features)
        + len(result.warning_features)
        + len(result.failed_features)
        == result.summary_statistics["total_features"]
    )
    
    # Verify average stability is in valid range
    assert 0 < result.summary_statistics["average_stability"] <= 1
    
    print(f"  PASS: {len(result.passed_features)}")
    print(f"  WARNING: {len(result.warning_features)}")
    print(f"  FAIL: {len(result.failed_features)}")
    print(f"  Avg Stability: {result.summary_statistics['average_stability']:.3f}")
    print(f"  Decisions: {decisions}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 9: Statistics engine."""
    print("\n" + "=" * 60)
    print("TEST 9: Statistics Engine")
    print("=" * 60)
    
    df = create_test_data()
    engine = FeatureStabilityEngine(window=50)
    stats_engine = FeatureStabilityStatisticsEngine()
    
    result = engine.compute(df)
    stats = stats_engine.analyze(result)
    
    assert stats.total_features == result.summary_statistics['total_features']

    print(f"  Total: {stats.total_features}")
    print(f"  PASS: {stats.passed_count}")
    print(f"  FAIL: {stats.failed_count}")
    print("  ✓ Statistics test passed")
    return True
